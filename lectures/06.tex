\section*{第 6 课}

% 第 1 张板书：积分定义、缩放及上半圆围道。
\paragraph{例 1.}
求实积分
\[
  I_{r,n}=\int_0^{+\infty}\frac{dx}{x^{2n}+r^{2n}},
  \qquad r>0,\quad n\in\mathbb{N}^{+}.
\]

\paragraph{解.}
作变量代换 $t=x/r$，得
\begin{align*}
  I_{r,n}
  &=\frac{1}{r^{2n}}\int_0^{+\infty}
    \frac{dx}{1+(x/r)^{2n}}\\
  &=\frac{1}{r^{2n-1}}\int_0^{+\infty}
    \frac{dt}{1+t^{2n}}
   =\frac{1}{r^{2n-1}}I_{1,n}.
\end{align*}
因此只需求 $I_{1,n}$。

令 $R>1$，取上半平面内的正向闭围道
\[
  \Gamma_R=[-R,R]\cup C_R,\qquad
  C_R:\ z=Re^{i\varphi},\quad 0\le\varphi\le\pi.
\]
由留数定理，
\begin{align*}
  \oint_{\Gamma_R}\frac{dz}{1+z^{2n}}
  &=\int_{-R}^{R}\frac{dx}{1+x^{2n}}
    +\int_{C_R}\frac{dz}{1+z^{2n}}\\
  &=2\pi i\sum_{k=1}^{n}
    \operatorname{Res}\left(\frac{1}{1+z^{2n}},z_k\right),
  \tag{*}
\end{align*}
其中 $z_1,\ldots,z_n$ 为上半平面内的极点。

% 第 2 张板书：大圆弧消失、极点位置及留数求和。
分子次数为 $0$，分母次数为 $2n\ge2$（$n\ge1$），因此当
$R\to+\infty$ 时，上半圆弧上的积分趋于 $0$。具体地，
\[
  \left|\int_{C_R}\frac{dz}{1+z^{2n}}\right|
  \le\frac{\pi R}{R^{2n}-1}\xrightarrow[R\to+\infty]{}0.
\]

在式 $(*)$ 中，极点满足
\[
  z^{2n}=-1=e^{(2k-1)\pi i}.
\]
取上半平面内的 $n$ 个根：
\[
  z_k=e^{\frac{(2k-1)\pi i}{2n}},
  \qquad k=1,2,\ldots,n.
\]
它们均为一阶极点，所以
\[
  \operatorname{Res}\left(\frac{1}{1+z^{2n}},z_k\right)
  =\frac{1}{2n z_k^{2n-1}}.
\]
由于 $z_k^{2n}=-1$，有
\[
  \frac{1}{z_k^{2n-1}}=-z_k.
\]
令 $R\to+\infty$，从式 $(*)$ 得
\begin{align*}
  2I_{1,n}
  &=2\pi i\sum_{k=1}^{n}\frac{1}{2n z_k^{2n-1}}\\
  &=\frac{\pi i}{n}\sum_{k=1}^{n}\frac{1}{z_k^{2n-1}}
   =-\frac{\pi i}{n}\sum_{k=1}^{n}z_k.
\end{align*}

令
\[
  \theta=\frac{\pi}{2n},\qquad
  q=e^{2i\theta}=e^{\pi i/n},
\]
则
\[
  z_k=\frac{q^k}{e^{i\theta}},
  \qquad q^n=e^{\pi i}=-1.
\]

% 第 3 张板书：等比数列求和及欧拉公式化简。
因此
\begin{align*}
  I_{1,n}
  &=-\frac{\pi i}{2n e^{i\theta}}\sum_{k=1}^{n}q^k\\
  &=-\frac{\pi i}{2n e^{i\theta}}\,
    \frac{q(1-q^n)}{1-q}\\
  &=-\frac{\pi i}{n}\,
    \frac{e^{i\theta}}{1-e^{2i\theta}}\\
  &=\frac{\pi i}{n}\,
    \frac{e^{i\theta}}{e^{2i\theta}-1}\\
  &=\frac{\pi}{2n\sin\theta},
\end{align*}
其中使用欧拉公式
\[
  \sin\theta=\frac{e^{i\theta}-e^{-i\theta}}{2i}
            =\frac{e^{2i\theta}-1}{2i e^{i\theta}}.
\]
最终
\[
  \boxed{
    I_{r,n}=\frac{\pi}
    {2n\,r^{2n-1}\sin\bigl(\frac{\pi}{2n}\bigr)}
  },\qquad r>0,\quad n\in\mathbb{N}^{+}.
\]

\paragraph{板书勘误（第 2--3 张）}
第 2 张的 $q^n$ 处写作 $+1$，应为 $-1$，因为
$q=e^{\pi i/n}$，从而 $q^n=e^{\pi i}=-1$。
另两张板书中的 $\theta$ 取值前后不一致；上文统一取
$\theta=\pi/(2n)$，并相应写为 $q=e^{2i\theta}$，
避免与 $q=e^{\pi i/n}$ 冲突。

% 此后继续追加第 6 课的后续板书，不另起导言区。


% 第 4 张板书：续前一例，用二倍角公式完成三角恒等式化简。
\paragraph{例 1 的另一种化简（续）}
仍令 $\theta=\pi/(2n)$。由上一页的等比数列求和结果，
\begin{align*}
  I_{1,n}
  &=-\frac{\pi i}{n}\frac{e^{i\theta}}{1-e^{2i\theta}}\\
  &=-\frac{\pi i}{n}
    \frac{e^{i\theta}}{1-\cos2\theta-i\sin2\theta}\\
  &=-\frac{\pi i}{n}
    \frac{e^{i\theta}}{2\sin^2\theta-2i\sin\theta\cos\theta}\\
  &=-\frac{\pi i}{2n}
    \frac{e^{i\theta}}{\sin\theta(\sin\theta-i\cos\theta)}\\
  &=-\frac{\pi i}{2n}
    \frac{e^{i\theta}}{(-i)(\cos\theta+i\sin\theta)\sin\theta}\\
  &=\frac{\pi}{2n}\frac{e^{i\theta}}{\sin\theta\,e^{i\theta}}
    =\frac{\pi}{2n\sin\theta}
    =\frac{\theta}{\sin\theta}
    =\frac{\pi}{2n\sin\frac{\pi}{2n}}.
\end{align*}
因此
\[
  I_{r,n}=\frac{1}{r^{2n-1}}I_{1,n}
  =\frac{\pi}{2nr^{2n-1}\sin\frac{\pi}{2n}}.
\]

% 第 4 张板书下半部：新例题、变量代换。
\paragraph{例 2.}
求实积分
\[
  J_{r,n}=\int_0^{+\infty}\frac{dx}{(x^2+r^2)^n},
  \qquad r>0,\quad n\in\mathbb N^{+}.
\]
令 $t=x/r$，则
\begin{align*}
  J_{r,n}
  &=\frac{1}{r^{2n}}
    \int_0^{+\infty}\frac{dx}{\bigl(1+(x/r)^2\bigr)^n}\\
  &=\frac{1}{r^{2n-1}}
    \int_0^{+\infty}\frac{dt}{(1+t^2)^n}
   =\frac{1}{r^{2n-1}}J_{1,n}.
\end{align*}
只需求
\[
  J_{1,n}=\int_0^{+\infty}\frac{dx}{(1+x^2)^n}.
\]

% 第 5 张板书：上半圆围道与 n 阶极点 i 的留数。
取上半圆正向闭围道
\[
  \Gamma_R=[-R,R]\cup C_R,\qquad R>1,\qquad
  C_R:\ z=Re^{i\varphi},\quad 0\le\varphi\le\pi.
\]
由留数定理，
\begin{align*}
  \oint_{\Gamma_R}\frac{dz}{(1+z^2)^n}
  &=\int_{-R}^{R}\frac{dx}{(1+x^2)^n}
    +\int_{C_R}\frac{dz}{(1+z^2)^n}\\
  &=2\pi i\,\operatorname{Res}
    \left(\frac{1}{(1+z^2)^n},i\right).
\end{align*}
当 $R\to+\infty$ 时，
\[
  \int_{C_R}\frac{dz}{(1+z^2)^n}\longrightarrow0.
\]
于是
\[
  \int_{-\infty}^{+\infty}\frac{dx}{(1+x^2)^n}
  =2\pi i\,\operatorname{Res}
   \left(\frac{1}{(1+z^2)^n},i\right)
  =2J_{1,n},
\]
即
\[
  J_{1,n}=\pi i\,\operatorname{Res}
  \left(\frac{1}{(1+z^2)^n},i\right).
\]

$\dfrac{1}{(1+z^2)^n}$ 在 $z=i$ 处有 $n$ 阶极点，
故要求其 Laurent 展开式中 $(z-i)^{-1}$ 的系数 $C_{-1}$。
将原函数写成
\begin{align*}
  \frac{1}{(1+z^2)^n}
  &=\frac{1}{(z-i)^n(z+i)^n}\\
  &=\frac{1}{(z-i)^n}
    \frac{1}{\bigl(z-(i-2i)\bigr)^n}\\
  &=\frac{1}{(z-i)^n}\cdot\frac{1}{(2i)^n}
    \cdot\frac{1}{\left(1+\dfrac{z-i}{2i}\right)^n}.
\end{align*}
所以只需求 $(1+X)^{-n}$ 展开式中 $X^{n-1}$ 的系数，
其中 $X=(z-i)/(2i)$。

% 第 6 张板书：广义二项式展开与负整数次幂系数。
\paragraph{引理（广义二项式展开）}
当 $|X|<1$ 时，
\[
  (1+X)^\alpha
  =1+\sum_{k=1}^{+\infty}
    \frac{\alpha(\alpha-1)\cdots(\alpha-k+1)}{k!}\,X^k.
\]
令 $\alpha=-n$，其中 $n\in\mathbb N^{+}$，得
\begin{align*}
  \frac{1}{(1+X)^n}
  &=1+\sum_{k=1}^{+\infty}
    \frac{(-n)(-n-1)\cdots(-n-k+1)}{k!}\,X^k\\
  &=1+\sum_{k=1}^{+\infty}
    (-1)^k\frac{(n+k-1)!}{k!(n-1)!}\,X^k\\
  &=1+\sum_{k=1}^{+\infty}
    (-1)^k\binom{n+k-1}{n-1}X^k\\
  &=\sum_{k=0}^{+\infty}
    (-1)^k\binom{n+k-1}{n-1}X^k.
\end{align*}
% 第 6 张板书到此为止；第 7 张继续求 C_{-1}。

% 第 7 张板书：提取 X^{n-1} 项并求 z=i 处的留数。
由上一张板书
\[
  \frac{1}{(1+X)^n}
  =\sum_{k=0}^{+\infty}(-1)^k
    \binom{n+k-1}{n-1}X^k,
\]
知 $X^{n-1}$ 项的系数为
\begin{align*}
  (-1)^{n-1}\binom{2n-2}{n-1}
  &=(-1)^{n-1}\binom{2(n-1)}{n-1}\\
  &=(-1)^{n-1}\frac{[2(n-1)]!}{[(n-1)!]^2}.
\end{align*}

又
\begin{align*}
  \frac{1}{(1+z^2)^n}
  &=\frac{1}{(z-i)^n}\,
    \frac{1}{\bigl(z-i+2i\bigr)^n}\\
  &=\frac{1}{(z-i)^n}\,
    \frac{1}{(2i)^n}\,
    \frac{1}{\left(1+\dfrac{z-i}{2i}\right)^n}\\
  &=\frac{1}{2^n i^n}\,
    \frac{1}{(z-i)^n}\,
    \sum_{k=0}^{+\infty}(-1)^k
    \binom{n+k-1}{n-1}X^k,
    \qquad X=\frac{z-i}{2i}.
\end{align*}
由于
\[
  X^{n-1}=\frac{(z-i)^{n-1}}{(2i)^{n-1}},
\]
故其与 $(z-i)^{-n}$ 相乘恰好产生 $(z-i)^{-1}$ 项。因此
\begin{align*}
  \operatorname{Res}
  \left(\frac{1}{(1+z^2)^n},i\right)
  &=\frac{1}{2^n i^n}\,
    (-1)^{n-1}\binom{2(n-1)}{n-1}
    \frac{1}{(2i)^{n-1}}\\
  &=\frac{(-1)^{n-1}}{2^{2n-1}i^{2n-1}}
    \binom{2(n-1)}{n-1}\\
  &=\frac{(-1)^{n-1}(-1)^n i}{2^{2n-1}}
    \binom{2(n-1)}{n-1}\\
  &=-\frac{i}{2^{2n-1}}\binom{2(n-1)}{n-1}.
\end{align*}

% 本张板书止于留数的计算；第 8 张板书继续代回 J_{1,n}。


% 第 8 张板书：代回留数、给出积分结果及 n 趋于无穷的讨论。
由上张板书计算所得
\[
  \operatorname{Res}\left(\frac{1}{(1+z^2)^n},i\right)
  =-\frac{i}{2^{2n-1}}\binom{2(n-1)}{n-1},
\]
故
\begin{align*}
  J_{1,n}
  &=\pi i\,\operatorname{Res}
    \left(\frac{1}{(1+z^2)^n},i\right)\\
  &=\frac{\pi i(-i)}{2^{2n-1}}\binom{2(n-1)}{n-1}\\
  &=\frac{\pi}{2^{2n-1}}\binom{2(n-1)}{n-1}\\
  &=\frac{\pi[2(n-1)]!}{2^{2n-1}[(n-1)!]^2}.
\end{align*}
所以
\[
  \boxed{
    J_{r,n}=\frac{1}{r^{2n-1}}J_{1,n}
    =\frac{\pi[2(n-1)]!}{(2r)^{2n-1}[(n-1)!]^2}
  },\qquad r>0,\quad n\in\mathbb N^+.
\]

\paragraph{关于 $n\to+\infty$ 的讨论}
由上式可得
\[
  \frac{J_{r,n+1}}{J_{r,n}}
  =\frac{1}{r^2}\left(1-\frac{1}{2n}\right)<1,
  \qquad r\ge1.
\]

先看 $r=1$。任取 $\varepsilon>0$，将积分拆为
\[
  J_{1,n}
  =\int_0^\varepsilon\frac{dx}{(1+x^2)^n}
   +\int_\varepsilon^{+\infty}\frac{dx}{(1+x^2)^n}.
\]
其中
\[
  0\le \int_0^\varepsilon\frac{dx}{(1+x^2)^n}
  \le\int_0^\varepsilon dx=\varepsilon,
\]
而
\[
  0\le\int_\varepsilon^{+\infty}\frac{dx}{(1+x^2)^n}
  \le\frac{1}{(1+\varepsilon^2)^{n-1}}
     \int_\varepsilon^{+\infty}\frac{dx}{1+x^2}
  \xrightarrow[n\to+\infty]{}0.
\]
因此
\[
  0\le\limsup_{n\to+\infty}J_{1,n}\le\varepsilon.
\]
令 $\varepsilon\to0^+$，得到
\[
  \boxed{\lim_{n\to+\infty}J_{1,n}=0}.
\]

另一方面，对 $r\ge1$，由上述比值，
\begin{align*}
  \frac{J_{r,n+1}}{J_{r,n}}\,
  \frac{J_{r,n}}{J_{r,n-1}}\cdots
  \frac{J_{r,2}}{J_{r,1}}
  &=\frac{J_{r,n+1}}{J_{r,1}}\\
  &=\prod_{k=1}^{n}\frac{1}{r^2}
       \left(1-\frac{1}{2k}\right)\\
  &\le \exp\left(-\frac12\sum_{k=1}^{n}\frac1k\right)
    \longrightarrow0.
\end{align*}
从而
\[
  \boxed{J_{r,n+1}\longrightarrow0\quad(n\to+\infty,\ r\ge1).}
\]

\paragraph{板书勘误（第 8 张）}
比值 $\dfrac{J_{r,n+1}}{J_{r,n}}$ 中，
板书的 $r$ 次幂写成 $r^{2n}$，应改为 $r^2$：
由 $J_{r,n}\propto r^{-(2n-1)}$ 可知相邻两项相除只相差 $r^{-2}$。
此外，分段估计极限时应先固定 $\varepsilon>0$，
再令 $n\to+\infty$，最后令 $\varepsilon\to0^+$；
上文已将这一步写为严格的不等式。

\section{共形映射（Analytic Mappings）}
\subsection*{保角映射（Angle-preserving Mappings）}

设 $w=f(z)$ 在 $z_0$ 的邻域内解析，且 $f'(z_0)\ne0$。
复数的极坐标表示为
\[
  f(z)\ne0\quad\Longrightarrow\quad
  f(z)=|f(z)|e^{i\varphi}.
\]
下面考察 $f$ 对相交曲线夹角的影响。

设平面上的两条光滑曲线 $C_1,C_2$ 在 $z_0$ 相交，
分别参数化为
\[
  z_1(t)=x_1(t)+iy_1(t),\qquad
  z_2(t)=x_2(t)+iy_2(t),
  \qquad z_1(t_0)=z_2(t_0)=z_0,
\]
并且
\[
  z_1'(t_0)=x_1'(t_0)+iy_1'(t_0)\ne0,\qquad
  z_2'(t_0)=x_2'(t_0)+iy_2'(t_0)\ne0.
\]
记两条曲线在 $z_0$ 的有向夹角为 $\theta$。
若写成
\[
  z_1'(t_0)=|z_1'(t_0)|e^{i\theta_1},\qquad
  z_2'(t_0)=|z_2'(t_0)|e^{i\theta_2},
\]
则
\[
  \theta=\theta_2-\theta_1
  =\arg z_2'(t_0)-\arg z_1'(t_0)
  =\arg\frac{z_2'(t_0)}{z_1'(t_0)}
  \pmod{2\pi}.
\]

在映射 $w=f(z)$ 下，两条曲线的像分别记为
\[
  w_1(t)=f(z_1(t)),\qquad w_2(t)=f(z_2(t)).
\]
由复合函数求导法则，像曲线在公共点 $f(z_0)$ 处的切向量为
\[
  w_1'(t_0)=f'(z_0)z_1'(t_0),\qquad
  w_2'(t_0)=f'(z_0)z_2'(t_0).
\]
由于 $f'(z_0)\ne0$，
\[
  \frac{w_2'(t_0)}{w_1'(t_0)}
  =\frac{f'(z_0)z_2'(t_0)}{f'(z_0)z_1'(t_0)}
  =\frac{z_2'(t_0)}{z_1'(t_0)}.
\]
故像曲线的有向夹角 $\varphi$ 满足
\[
  \varphi
  =\arg w_2'(t_0)-\arg w_1'(t_0)
  =\arg z_2'(t_0)-\arg z_1'(t_0)
  =\theta\pmod{2\pi}.
\]
因此 $f$ 在 $f'(z_0)\ne0$ 的点处保角。

% 第 10 张板书：保角结论的续写与五个例子。
\subsection*{例子}
\begin{enumerate}
  \item \textbf{多项式映射}
  \[
    f(z)=P_n(z),\qquad f'(z)=P_n'(z).
  \]
  其导数仍为多项式；在 $P_n'(z)\ne0$ 的点处保角。

  \item \textbf{正弦映射}
  \[
    f(z)=\sin z,\qquad f'(z)=\cos z.
  \]
  在 $\cos z\ne0$ 的点处保角；当 $\cos z=0$ 时，
  上述保角判据不适用。

  \item \textbf{常值映射}
  \[
    f(z)=C,\qquad f'(z)=0.
  \]
  常值映射不保角。

  \item \textbf{指数映射}
  \[
    f(z)=e^z,\qquad f'(z)=e^z\ne0.
  \]
  因此在所有有限复数点处保角。

  \item \textbf{分式线性变换（Möbius Mapping）}
  \[
    f(z)=\frac{az+b}{cz+d},\qquad
    a,b,c,d\in\mathbb C,\qquad ad-bc\ne0,
  \]
  记
  \[
    A=\begin{pmatrix}a&b\\c&d\end{pmatrix},
    \qquad \det A=ad-bc\ne0.
  \]
  在复平面中上述分式的定义要求 $cz+d\ne0$。
  % 老师右侧关于扩充复平面/无穷远点的板书较模糊，待后续照片核对。
\end{enumerate}
