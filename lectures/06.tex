\section*{第 6 课}

% 第 1 张板书：积分定义、缩放及上半圆围道。
\paragraph{例 1.}
求实积分
\[
  I_{r,n}=\int_0^{+\infty}\frac{dx}{x^{2n}+r^{2n}},
  \qquad r>0,\quad n\in\mathbb{N}^{+}.
\]

\paragraph{解.}
作变量代换 $t=x/r$，得
\begin{align*}
  I_{r,n}
  &=\frac{1}{r^{2n}}\int_0^{+\infty}
    \frac{dx}{1+(x/r)^{2n}}\\
  &=\frac{1}{r^{2n-1}}\int_0^{+\infty}
    \frac{dt}{1+t^{2n}}
   =\frac{1}{r^{2n-1}}I_{1,n}.
\end{align*}
因此只需求 $I_{1,n}$。

令 $R>1$，取上半平面内的正向闭围道
\[
  \Gamma_R=[-R,R]\cup C_R,\qquad
  C_R:\ z=Re^{i\varphi},\quad 0\le\varphi\le\pi.
\]
由留数定理，
\begin{align*}
  \oint_{\Gamma_R}\frac{dz}{1+z^{2n}}
  &=\int_{-R}^{R}\frac{dx}{1+x^{2n}}
    +\int_{C_R}\frac{dz}{1+z^{2n}}\\
  &=2\pi i\sum_{k=1}^{n}
    \operatorname{Res}\left(\frac{1}{1+z^{2n}},z_k\right),
  \tag{*}
\end{align*}
其中 $z_1,\ldots,z_n$ 为上半平面内的极点。

% 第 2 张板书：大圆弧消失、极点位置及留数求和。
分子次数为 $0$，分母次数为 $2n\ge2$（$n\ge1$），因此当
$R\to+\infty$ 时，上半圆弧上的积分趋于 $0$。具体地，
\[
  \left|\int_{C_R}\frac{dz}{1+z^{2n}}\right|
  \le\frac{\pi R}{R^{2n}-1}\xrightarrow[R\to+\infty]{}0.
\]

在式 $(*)$ 中，极点满足
\[
  z^{2n}=-1=e^{(2k-1)\pi i}.
\]
取上半平面内的 $n$ 个根：
\[
  z_k=e^{\frac{(2k-1)\pi i}{2n}},
  \qquad k=1,2,\ldots,n.
\]
它们均为一阶极点，所以
\[
  \operatorname{Res}\left(\frac{1}{1+z^{2n}},z_k\right)
  =\frac{1}{2n z_k^{2n-1}}.
\]
由于 $z_k^{2n}=-1$，有
\[
  \frac{1}{z_k^{2n-1}}=-z_k.
\]
令 $R\to+\infty$，从式 $(*)$ 得
\begin{align*}
  2I_{1,n}
  &=2\pi i\sum_{k=1}^{n}\frac{1}{2n z_k^{2n-1}}\\
  &=\frac{\pi i}{n}\sum_{k=1}^{n}\frac{1}{z_k^{2n-1}}
   =-\frac{\pi i}{n}\sum_{k=1}^{n}z_k.
\end{align*}

令
\[
  \theta=\frac{\pi}{2n},\qquad
  q=e^{2i\theta}=e^{\pi i/n},
\]
则
\[
  z_k=\frac{q^k}{e^{i\theta}},
  \qquad q^n=e^{\pi i}=-1.
\]

% 第 3 张板书：等比数列求和及欧拉公式化简。
因此
\begin{align*}
  I_{1,n}
  &=-\frac{\pi i}{2n e^{i\theta}}\sum_{k=1}^{n}q^k\\
  &=-\frac{\pi i}{2n e^{i\theta}}\,
    \frac{q(1-q^n)}{1-q}\\
  &=-\frac{\pi i}{n}\,
    \frac{e^{i\theta}}{1-e^{2i\theta}}\\
  &=\frac{\pi i}{n}\,
    \frac{e^{i\theta}}{e^{2i\theta}-1}\\
  &=\frac{\pi}{2n\sin\theta},
\end{align*}
其中使用欧拉公式
\[
  \sin\theta=\frac{e^{i\theta}-e^{-i\theta}}{2i}
            =\frac{e^{2i\theta}-1}{2i e^{i\theta}}.
\]
最终
\[
  \boxed{
    I_{r,n}=\frac{\pi}
    {2n\,r^{2n-1}\sin\bigl(\frac{\pi}{2n}\bigr)}
  },\qquad r>0,\quad n\in\mathbb{N}^{+}.
\]

\paragraph{板书勘误（第 2--3 张）}
第 2 张的 $q^n$ 处写作 $+1$，应为 $-1$，因为
$q=e^{\pi i/n}$，从而 $q^n=e^{\pi i}=-1$。
另两张板书中的 $\theta$ 取值前后不一致；上文统一取
$\theta=\pi/(2n)$，并相应写为 $q=e^{2i\theta}$，
避免与 $q=e^{\pi i/n}$ 冲突。

% 此后继续追加第 6 课的后续板书，不另起导言区。


% 第 4 张板书：续前一例，用二倍角公式完成三角恒等式化简。
\paragraph{例 1 的另一种化简（续）}
仍令 $\theta=\pi/(2n)$。由上一页的等比数列求和结果，
\begin{align*}
  I_{1,n}
  &=-\frac{\pi i}{n}\frac{e^{i\theta}}{1-e^{2i\theta}}\\
  &=-\frac{\pi i}{n}
    \frac{e^{i\theta}}{1-\cos2\theta-i\sin2\theta}\\
  &=-\frac{\pi i}{n}
    \frac{e^{i\theta}}{2\sin^2\theta-2i\sin\theta\cos\theta}\\
  &=-\frac{\pi i}{2n}
    \frac{e^{i\theta}}{\sin\theta(\sin\theta-i\cos\theta)}\\
  &=-\frac{\pi i}{2n}
    \frac{e^{i\theta}}{(-i)(\cos\theta+i\sin\theta)\sin\theta}\\
  &=\frac{\pi}{2n}\frac{e^{i\theta}}{\sin\theta\,e^{i\theta}}
    =\frac{\pi}{2n\sin\theta}
    =\frac{\theta}{\sin\theta}
    =\frac{\pi}{2n\sin\frac{\pi}{2n}}.
\end{align*}
因此
\[
  I_{r,n}=\frac{1}{r^{2n-1}}I_{1,n}
  =\frac{\pi}{2nr^{2n-1}\sin\frac{\pi}{2n}}.
\]

% 第 4 张板书下半部：新例题、变量代换。
\paragraph{例 2.}
求实积分
\[
  J_{r,n}=\int_0^{+\infty}\frac{dx}{(x^2+r^2)^n},
  \qquad r>0,\quad n\in\mathbb N^{+}.
\]
令 $t=x/r$，则
\begin{align*}
  J_{r,n}
  &=\frac{1}{r^{2n}}
    \int_0^{+\infty}\frac{dx}{\bigl(1+(x/r)^2\bigr)^n}\\
  &=\frac{1}{r^{2n-1}}
    \int_0^{+\infty}\frac{dt}{(1+t^2)^n}
   =\frac{1}{r^{2n-1}}J_{1,n}.
\end{align*}
只需求
\[
  J_{1,n}=\int_0^{+\infty}\frac{dx}{(1+x^2)^n}.
\]

% 第 5 张板书：上半圆围道与 n 阶极点 i 的留数。
取上半圆正向闭围道
\[
  \Gamma_R=[-R,R]\cup C_R,\qquad R>1,\qquad
  C_R:\ z=Re^{i\varphi},\quad 0\le\varphi\le\pi.
\]
由留数定理，
\begin{align*}
  \oint_{\Gamma_R}\frac{dz}{(1+z^2)^n}
  &=\int_{-R}^{R}\frac{dx}{(1+x^2)^n}
    +\int_{C_R}\frac{dz}{(1+z^2)^n}\\
  &=2\pi i\,\operatorname{Res}
    \left(\frac{1}{(1+z^2)^n},i\right).
\end{align*}
当 $R\to+\infty$ 时，
\[
  \int_{C_R}\frac{dz}{(1+z^2)^n}\longrightarrow0.
\]
于是
\[
  \int_{-\infty}^{+\infty}\frac{dx}{(1+x^2)^n}
  =2\pi i\,\operatorname{Res}
   \left(\frac{1}{(1+z^2)^n},i\right)
  =2J_{1,n},
\]
即
\[
  J_{1,n}=\pi i\,\operatorname{Res}
  \left(\frac{1}{(1+z^2)^n},i\right).
\]

$\dfrac{1}{(1+z^2)^n}$ 在 $z=i$ 处有 $n$ 阶极点，
故要求其 Laurent 展开式中 $(z-i)^{-1}$ 的系数 $C_{-1}$。
将原函数写成
\begin{align*}
  \frac{1}{(1+z^2)^n}
  &=\frac{1}{(z-i)^n(z+i)^n}\\
  &=\frac{1}{(z-i)^n}
    \frac{1}{\bigl(z-(i-2i)\bigr)^n}\\
  &=\frac{1}{(z-i)^n}\cdot\frac{1}{(2i)^n}
    \cdot\frac{1}{\left(1+\dfrac{z-i}{2i}\right)^n}.
\end{align*}
所以只需求 $(1+X)^{-n}$ 展开式中 $X^{n-1}$ 的系数，
其中 $X=(z-i)/(2i)$。

% 第 6 张板书：广义二项式展开与负整数次幂系数。
\paragraph{引理（广义二项式展开）}
当 $|X|<1$ 时，
\[
  (1+X)^\alpha
  =1+\sum_{k=1}^{+\infty}
    \frac{\alpha(\alpha-1)\cdots(\alpha-k+1)}{k!}\,X^k.
\]
令 $\alpha=-n$，其中 $n\in\mathbb N^{+}$，得
\begin{align*}
  \frac{1}{(1+X)^n}
  &=1+\sum_{k=1}^{+\infty}
    \frac{(-n)(-n-1)\cdots(-n-k+1)}{k!}\,X^k\\
  &=1+\sum_{k=1}^{+\infty}
    (-1)^k\frac{(n+k-1)!}{k!(n-1)!}\,X^k\\
  &=1+\sum_{k=1}^{+\infty}
    (-1)^k\binom{n+k-1}{n-1}X^k\\
  &=\sum_{k=0}^{+\infty}
    (-1)^k\binom{n+k-1}{n-1}X^k.
\end{align*}
% 第 6 张板书到此为止；第 7 张继续求 C_{-1}。

% 第 7 张板书：提取 X^{n-1} 项并求 z=i 处的留数。
由上一张板书
\[
  \frac{1}{(1+X)^n}
  =\sum_{k=0}^{+\infty}(-1)^k
    \binom{n+k-1}{n-1}X^k,
\]
知 $X^{n-1}$ 项的系数为
\begin{align*}
  (-1)^{n-1}\binom{2n-2}{n-1}
  &=(-1)^{n-1}\binom{2(n-1)}{n-1}\\
  &=(-1)^{n-1}\frac{[2(n-1)]!}{[(n-1)!]^2}.
\end{align*}

又
\begin{align*}
  \frac{1}{(1+z^2)^n}
  &=\frac{1}{(z-i)^n}\,
    \frac{1}{\bigl(z-i+2i\bigr)^n}\\
  &=\frac{1}{(z-i)^n}\,
    \frac{1}{(2i)^n}\,
    \frac{1}{\left(1+\dfrac{z-i}{2i}\right)^n}\\
  &=\frac{1}{2^n i^n}\,
    \frac{1}{(z-i)^n}\,
    \sum_{k=0}^{+\infty}(-1)^k
    \binom{n+k-1}{n-1}X^k,
    \qquad X=\frac{z-i}{2i}.
\end{align*}
由于
\[
  X^{n-1}=\frac{(z-i)^{n-1}}{(2i)^{n-1}},
\]
故其与 $(z-i)^{-n}$ 相乘恰好产生 $(z-i)^{-1}$ 项。因此
\begin{align*}
  \operatorname{Res}
  \left(\frac{1}{(1+z^2)^n},i\right)
  &=\frac{1}{2^n i^n}\,
    (-1)^{n-1}\binom{2(n-1)}{n-1}
    \frac{1}{(2i)^{n-1}}\\
  &=\frac{(-1)^{n-1}}{2^{2n-1}i^{2n-1}}
    \binom{2(n-1)}{n-1}\\
  &=\frac{(-1)^{n-1}(-1)^n i}{2^{2n-1}}
    \binom{2(n-1)}{n-1}\\
  &=-\frac{i}{2^{2n-1}}\binom{2(n-1)}{n-1}.
\end{align*}

% 本张板书止于留数的计算；第 8 张板书继续代回 J_{1,n}。


% 第 8 张板书：代回留数、给出积分结果及 n 趋于无穷的讨论。
由上张板书计算所得
\[
  \operatorname{Res}\left(\frac{1}{(1+z^2)^n},i\right)
  =-\frac{i}{2^{2n-1}}\binom{2(n-1)}{n-1},
\]
故
\begin{align*}
  J_{1,n}
  &=\pi i\,\operatorname{Res}
    \left(\frac{1}{(1+z^2)^n},i\right)\\
  &=\frac{\pi i(-i)}{2^{2n-1}}\binom{2(n-1)}{n-1}\\
  &=\frac{\pi}{2^{2n-1}}\binom{2(n-1)}{n-1}\\
  &=\frac{\pi[2(n-1)]!}{2^{2n-1}[(n-1)!]^2}.
\end{align*}
所以
\[
  \boxed{
    J_{r,n}=\frac{1}{r^{2n-1}}J_{1,n}
    =\frac{\pi[2(n-1)]!}{(2r)^{2n-1}[(n-1)!]^2}
  },\qquad r>0,\quad n\in\mathbb N^+.
\]

\paragraph{关于 $n\to+\infty$ 的讨论}
由上式可得
\[
  \frac{J_{r,n+1}}{J_{r,n}}
  =\frac{1}{r^2}\left(1-\frac{1}{2n}\right)<1,
  \qquad r\ge1.
\]

先看 $r=1$。任取 $\varepsilon>0$，将积分拆为
\[
  J_{1,n}
  =\int_0^\varepsilon\frac{dx}{(1+x^2)^n}
   +\int_\varepsilon^{+\infty}\frac{dx}{(1+x^2)^n}.
\]
其中
\[
  0\le \int_0^\varepsilon\frac{dx}{(1+x^2)^n}
  \le\int_0^\varepsilon dx=\varepsilon,
\]
而
\[
  0\le\int_\varepsilon^{+\infty}\frac{dx}{(1+x^2)^n}
  \le\frac{1}{(1+\varepsilon^2)^{n-1}}
     \int_\varepsilon^{+\infty}\frac{dx}{1+x^2}
  \xrightarrow[n\to+\infty]{}0.
\]
因此
\[
  0\le\limsup_{n\to+\infty}J_{1,n}\le\varepsilon.
\]
令 $\varepsilon\to0^+$，得到
\[
  \boxed{\lim_{n\to+\infty}J_{1,n}=0}.
\]

另一方面，对 $r\ge1$，由上述比值，
\begin{align*}
  \frac{J_{r,n+1}}{J_{r,n}}\,
  \frac{J_{r,n}}{J_{r,n-1}}\cdots
  \frac{J_{r,2}}{J_{r,1}}
  &=\frac{J_{r,n+1}}{J_{r,1}}\\
  &=\prod_{k=1}^{n}\frac{1}{r^2}
       \left(1-\frac{1}{2k}\right)\\
  &\le \exp\left(-\frac12\sum_{k=1}^{n}\frac1k\right)
    \longrightarrow0.
\end{align*}
从而
\[
  \boxed{J_{r,n+1}\longrightarrow0\quad(n\to+\infty,\ r\ge1).}
\]

\paragraph{板书勘误（第 8 张）}
比值 $\dfrac{J_{r,n+1}}{J_{r,n}}$ 中，
板书的 $r$ 次幂写成 $r^{2n}$，应改为 $r^2$：
由 $J_{r,n}\propto r^{-(2n-1)}$ 可知相邻两项相除只相差 $r^{-2}$。
此外，分段估计极限时应先固定 $\varepsilon>0$，
再令 $n\to+\infty$，最后令 $\varepsilon\to0^+$；
上文已将这一步写为严格的不等式。

% 第 9 张板书：第六章开篇——解析映射的保角性。
\setcounter{section}{5}
\section{共形映射（Analytic Mappings）}
\subsection*{保角映射（Angle-preserving Mappings）}

设 $w=f(z)$ 在 $z_0$ 的邻域内解析，且 $f'(z_0)\ne0$。
复数的极坐标表示为
\[
  f(z)\ne0\quad\Longrightarrow\quad
  f(z)=|f(z)|e^{i\varphi}.
\]
下面考察 $f$ 对相交曲线夹角的影响。

设平面上的两条光滑曲线 $C_1,C_2$ 在 $z_0$ 相交，
分别参数化为
\[
  z_1(t)=x_1(t)+iy_1(t),\qquad
  z_2(t)=x_2(t)+iy_2(t),
  \qquad z_1(t_0)=z_2(t_0)=z_0,
\]
并且
\[
  z_1'(t_0)=x_1'(t_0)+iy_1'(t_0)\ne0,\qquad
  z_2'(t_0)=x_2'(t_0)+iy_2'(t_0)\ne0.
\]
记两条曲线在 $z_0$ 的有向夹角为 $\theta$。
若写成
\[
  z_1'(t_0)=|z_1'(t_0)|e^{i\theta_1},\qquad
  z_2'(t_0)=|z_2'(t_0)|e^{i\theta_2},
\]
则
\[
  \theta=\theta_2-\theta_1
  =\arg z_2'(t_0)-\arg z_1'(t_0)
  =\arg\frac{z_2'(t_0)}{z_1'(t_0)}
  \pmod{2\pi}.
\]

在映射 $w=f(z)$ 下，两条曲线的像分别记为
\[
  w_1(t)=f(z_1(t)),\qquad w_2(t)=f(z_2(t)).
\]
由复合函数求导法则，像曲线在公共点 $f(z_0)$ 处的切向量为
\[
  w_1'(t_0)=f'(z_0)z_1'(t_0),\qquad
  w_2'(t_0)=f'(z_0)z_2'(t_0).
\]
由于 $f'(z_0)\ne0$，
\[
  \frac{w_2'(t_0)}{w_1'(t_0)}
  =\frac{f'(z_0)z_2'(t_0)}{f'(z_0)z_1'(t_0)}
  =\frac{z_2'(t_0)}{z_1'(t_0)}.
\]
故像曲线的有向夹角 $\varphi$ 满足
\[
  \varphi
  =\arg w_2'(t_0)-\arg w_1'(t_0)
  =\arg z_2'(t_0)-\arg z_1'(t_0)
  =\theta\pmod{2\pi}.
\]
因此 $f$ 在 $f'(z_0)\ne0$ 的点处保角。

% 第 10 张板书：保角结论的续写与五个例子。
\subsection*{例子}
\begin{enumerate}
  \item \textbf{多项式映射}
  \[
    f(z)=P_n(z),\qquad f'(z)=P_n'(z).
  \]
  其导数仍为多项式；在 $P_n'(z)\ne0$ 的点处保角。

  \item \textbf{正弦映射}
  \[
    f(z)=\sin z,\qquad f'(z)=\cos z.
  \]
  在 $\cos z\ne0$ 的点处保角；当 $\cos z=0$ 时，
  上述保角判据不适用。

  \item \textbf{常值映射}
  \[
    f(z)=C,\qquad f'(z)=0.
  \]
  常值映射不保角。

  \item \textbf{指数映射}
  \[
    f(z)=e^z,\qquad f'(z)=e^z\ne0.
  \]
  因此在所有有限复数点处保角。

  \item \textbf{分式线性变换（Möbius Mapping）}
  \[
    f(z)=\frac{az+b}{cz+d},\qquad
    a,b,c,d\in\mathbb C,\qquad ad-bc\ne0,
  \]
  记
  \[
    A=\begin{pmatrix}a&b\\c&d\end{pmatrix},
    \qquad \det A=ad-bc\ne0.
  \]
  在复平面中上述分式的定义要求 $cz+d\ne0$。
  % 老师右侧关于扩充复平面/无穷远点的板书较模糊，待后续照片核对。
\end{enumerate}

% 第 11 张板书：Möbius 变换的导数与退化情形。
\subsection*{Möbius 变换的矩阵表示（续）}
设
\[
  w=\frac{az+b}{cz+d},\qquad
  A=\begin{pmatrix}a&b\\c&d\end{pmatrix},
  \qquad a,b,c,d\in\mathbb C.
\]
在 $cz+d\ne0$ 处，
\[
  \frac{dw}{dz}=\frac{ad-bc}{(cz+d)^2}.
\]
若 $\det A=ad-bc\ne0$，则 $dw/dz\ne0$，因此在其定义域内保角。

若 $ad-bc=0$，则在定义域内 $w$ 为常数。
当 $c,d\ne0$ 时，由板书的两种写法，
\[
  w\equiv w(0)=\frac bd,\qquad
  w=\frac{a+b/z}{c+d/z}\xrightarrow[z\to\infty]{}\frac ac,
  \qquad\therefore\quad\frac ac=\frac bd.
\]

% 第 12 张板书：矩阵的比例等价、归一化及逆变换。
\paragraph{矩阵归一化}
对任意 $\lambda\ne0$，
\[
  \frac{az+b}{cz+d}
  =\frac{\lambda az+\lambda b}{\lambda cz+\lambda d}.
\]
故 $A$ 与 $\lambda A$ 表示同一分式线性变换。可取
\[
  \det(\lambda A)=\lambda^2\det A=1
  \quad\Longrightarrow\quad
  \lambda^2=\frac1{ad-bc},\qquad
  \lambda=\pm\frac1{\sqrt{ad-bc}}.
\]
记
\[
  A'=\lambda A
  =\begin{pmatrix}a'&b'\\c'&d'\end{pmatrix},
\]
其中
\[
  a'=\frac{\pm a}{\sqrt{ad-bc}},\quad
  b'=\frac{\pm b}{\sqrt{ad-bc}},\quad
  c'=\frac{\pm c}{\sqrt{ad-bc}},\quad
  d'=\frac{\pm d}{\sqrt{ad-bc}}.
\]
于是
\[
  \det A'=a'd'-b'c'=\frac{ad-bc}{ad-bc}=1,
\]
即归一化得到 $\mathrm{SL}(2,\mathbb C)$ 中的矩阵。

\paragraph{逆变换}
由
\begin{align*}
  w=\frac{az+b}{cz+d}
  &\Longrightarrow(cz+d)w=az+b\\
  &\Longrightarrow(cw-a)z=b-dw\\
  &\Longrightarrow
  z=\frac{-dw+b}{cw-a}.
\end{align*}
因此逆 Möbius 变换可用矩阵
\[
  B=\begin{pmatrix}-d&b\\c&-a\end{pmatrix}
\]
表示，且 $\det B=ad-bc=\det A\ne0$。
通常意义下的矩阵逆则为
\[
  A^{-1}=\frac1{ad-bc}
  \begin{pmatrix}d&-b\\-c&a\end{pmatrix}
  =-\frac1{ad-bc}B.
\]

\paragraph{板书勘误（第 11--12 张）}
\begin{enumerate}
\item $\dfrac{dw}{dz}\ne0$ 只在 $cz+d\ne0$ 的定义域内成立；
若 $c\ne0$，原分式在 $z=-d/c$ 无定义。
\item 板书圈出的 $B$ 是\textbf{逆变换的表示矩阵}，
不能直接视作一般意义的 $A^{-1}$。
两者只差非零数倍，因而表示同一逆 Möbius 变换。
\end{enumerate}

% 第 12 张板书结束；第 13 张开始介绍 Möbius 变换的复合。


% 第 13 张板书：Möbius 变换的单位元、复合运算和群。
\paragraph{恒等变换及复合}
单位矩阵
\[
  I_2=\begin{pmatrix}1&0\\0&1\end{pmatrix}
\]
对应恒等变换 $w=z$（即 $a=d=1$，$b=c=0$）。

令
\[
  A_1=\begin{pmatrix}a_1&b_1\\c_1&d_1\end{pmatrix},
  \quad
  A_2=\begin{pmatrix}a_2&b_2\\c_2&d_2\end{pmatrix},
\]
分别对应
\[
  w_1(z)=\frac{a_1z+b_1}{c_1z+d_1},\qquad
  w_2(w)=\frac{a_2w+b_2}{c_2w+d_2}.
\]
先作 $w_1$，再作 $w_2$，得到
\begin{align*}
  (w_2\circ w_1)(z)
  &=\frac{a_2\dfrac{a_1z+b_1}{c_1z+d_1}+b_2}
          {c_2\dfrac{a_1z+b_1}{c_1z+d_1}+d_2}\\
  &=\frac{a_2(a_1z+b_1)+b_2(c_1z+d_1)}
          {c_2(a_1z+b_1)+d_2(c_1z+d_1)}\\
  &=\frac{(a_2a_1+b_2c_1)z+(a_2b_1+b_2d_1)}
          {(c_2a_1+d_2c_1)z+(c_2b_1+d_2d_1)}\\
  &=\frac{Az+B}{Cz+D},
\end{align*}
其中
\[
  \begin{pmatrix}A&B\\C&D\end{pmatrix}
  =A_2A_1
  =\begin{pmatrix}
    a_2a_1+b_2c_1&a_2b_1+b_2d_1\\
    c_2a_1+d_2c_1&c_2b_1+d_2d_1
   \end{pmatrix},
\]
且
\[
  AD-BC
  =\det(A_2A_1)
  =(a_2d_2-b_2c_2)(a_1d_1-b_1c_1)\ne0.
\]
故两个 Möbius 变换的复合仍为 Möbius 变换。

\paragraph{群结构}
记这些变换所构成的集合为 $G$。其复合运算满足
封闭性、结合律，并具有恒等元及逆元：
\[
  a,b\in G\Rightarrow a\circ b\in G,\qquad
  a\in G\Rightarrow a^{-1}\in G,\qquad
  \mathrm{id}\circ a=a\circ\mathrm{id}=a.
\]
一般不满足交换律：
\[
  A_1A_2\ne A_2A_1,\qquad
  w_1\circ w_2\ne w_2\circ w_1
  \quad\text{（一般情形）}.
\]
由于比例相差非零常数的矩阵对应相同变换，
此处 $G$ 可以用 $\mathrm{PGL}(2,\mathbb C)$ 表示。

% 第 14 张板书：不动点及其代数方程。
\paragraph{不动点（Fixed Point）}
若 $f(z_0)=z_0$，则称 $z_0$ 为 $f$ 的一个不动点。
对于
\[
  f(z)=\frac{az+b}{cz+d},\qquad ad-bc\ne0,
\]
有限不动点满足
\begin{align*}
  \frac{az+b}{cz+d}=z
  &\iff az+b=cz^2+dz\\
  &\iff cz^2+(d-a)z-b=0.
\end{align*}

\begin{enumerate}
  \item 若 $c\ne0$，上述方程为二次方程，判别式为
  \[
    \Delta=(d-a)^2+4bc.
  \]
  当 $\Delta\ne0$ 时有两个不同的不动点；
  当 $\Delta=0$ 时有一个二重不动点。
  \item 若 $c=0$，由于 $ad\ne0$，则
  \[
    (d-a)z=b.
  \]
  当 $d\ne a$ 时，有一个有限不动点
  \[
    z=\frac{b}{d-a}.
  \]
  当 $d=a$ 且 $b\ne0$ 时，没有有限不动点；
  当 $d=a$ 且 $b=0$ 时，$f(z)\equiv z$，
  每一点都是不动点。
\end{enumerate}

% 第 15 张板书：典型变换的不动点及几何联想。
\paragraph{不动点的几个例子}
\begin{enumerate}
  \item 平移变换
  \[
    w=z+\alpha,\qquad \alpha\ne0.
  \]
  若 $\alpha=a+ib$、$z=x+iy$、$w=x'+iy'$，则
  \[
    x'=x+a,\qquad y'=y+b.
  \]
  它在复平面没有有限不动点。
  \item 仿射变换
  \[
    w=az+b,\qquad a\ne0.
  \]
  当 $a\ne1$ 时，有限不动点为
  \[
    z=\frac{b}{1-a}.
  \]
  \item 一般分式线性变换
  \[
    w=\frac{az+b}{cz+d},\qquad
    c\ne0,\quad ad-bc\ne0,
  \]
  其有限不动点由上述二次方程确定。
  \item 恒等变换 $w\equiv z$：所有点均为不动点。
\end{enumerate}
在扩充复平面 $\widehat{\mathbb C}=\mathbb C\cup\{\infty\}$ 中，
$c=0$ 的 Möbius 变换还固定无穷远点；
因此非恒等 Möbius 变换在球面上有一个或两个不同的不动点。

\paragraph{几何补充（板书后半部分）}
板书由 Euclid Geometry（欧氏几何）转入球面与其他几何的讨论：
欧氏平面中的平行直线，以及三角形内角和
\[
  \alpha+\beta+\gamma=180^\circ=\pi.
\]
板书接着画出了球面上的曲线、大圆与两点间的连接，
讨论球面上的\textbf{测地线}：球面上的测地线沿大圆，
两点间较短的大圆弧在非对径情形下是最短路径。
右侧提及“罗巴切夫斯基几何”和“Gauss--B.”；
后者具体所指待后续板书确认。

\paragraph{板书勘误（第 13--15 张）}
\begin{enumerate}
  \item 复合 $w_2\circ w_1$ 对应的矩阵顺序是 $A_2A_1$，
  而不是 $A_1A_2$。
  \item 不动点方程的二次项应为 $cz^2$，
  不能写作 $(c-a)z^2$；相应的分情况条件是
  $c\ne0$ 或 $c=0$，判别式为
  $\Delta=(d-a)^2+4bc$。
  \item 对不动点个数应区分普通复平面与扩充复平面：
  平移变换在 $\mathbb C$ 内无不动点，
  但在 $\widehat{\mathbb C}$ 中固定 $\infty$。
\end{enumerate}

% 第 15 张板书结束；第 16 张继续介绍基本变换。


% 第 16 张板书：Möbius 变换的三种基本变换。
\paragraph{三种基本变换}
一般的 Möbius 变换为
\[
  w=\frac{az+b}{cz+d}.
\]
下面列出基本变换及其几何效果。

\begin{enumerate}
  \item \textbf{平移}
  \[
    w=z+\alpha,\qquad
    \alpha=a+ib\ne0.
  \]

  \item \textbf{伸缩与旋转}
  \[
    w=\alpha z,\qquad \alpha\ne0.
  \]
  设
  \[
    \alpha=|\alpha|e^{i\theta_\alpha},\qquad
    z=re^{i\theta},
  \]
  则
  \[
    w=|\alpha|r\,e^{i(\theta_\alpha+\theta)},
    \qquad |w|=|\alpha|\,|z|.
  \]
  因此，$|\alpha|>1$ 为放大，$0<|\alpha|<1$ 为缩小；
  $|\alpha|=1$ 时为旋转。特别地，
  \[
    |\alpha|=1\quad\Longrightarrow\quad
    \alpha=e^{i\theta_\alpha},\quad
    w=re^{i(\theta_\alpha+\theta)}.
  \]
  写 $z=x+iy$、$w=u+iv$，可用旋转矩阵表示为
  \[
    \begin{pmatrix}u\\v\end{pmatrix}
    =
    \underbrace{\begin{pmatrix}
      \cos\theta_\alpha&-\sin\theta_\alpha\\
      \sin\theta_\alpha&\cos\theta_\alpha
    \end{pmatrix}}_{A(\theta_\alpha)}
    \begin{pmatrix}x\\y\end{pmatrix}.
  \]
  且
  \[
    A^{-1}(\theta_\alpha)=A^{\mathsf T}(\theta_\alpha),
  \]
  即旋转矩阵为正交矩阵。

  \item \textbf{倒数变换}
  \[
    w=\frac1z,\qquad z\ne0.
  \]
  令
  \[
    z=re^{i\theta}\quad\Longrightarrow\quad
    w=\frac1r e^{-i\theta}.
  \]
  因此
  \[
    r>1\Longrightarrow |w|=\frac1r<1,\qquad
    r\to+\infty\Longrightarrow |w|\to0,\qquad
    r\to0^+\Longrightarrow |w|\to+\infty.
  \]
  也就是说，单位圆的内外互换，单位圆自身保持不变。
  同时
  \[
    \theta\in(0,\pi)
    \Longrightarrow
    -\theta\in(-\pi,0)\equiv(\pi,2\pi)\pmod{2\pi},
  \]
  即上半平面映到下半平面。
\end{enumerate}

\paragraph{板书勘误（第 16 张）}
\begin{enumerate}
  \item 板书在讨论 $w=\alpha z$ 时，将 $|w|$ 与含有
  $e^{i\theta}$ 的复数表达式连写。正确区分为
  $w=|\alpha|r e^{i(\theta_\alpha+\theta)}$ 与
  $|w|=|\alpha|r$。
  \item 旋转矩阵须与向量的行、列约定一致。
  板书采用行向量 $(x,y)$ 右乘
  $\left(\begin{smallmatrix}\cos\theta&-\sin\theta\\
  \sin\theta&\cos\theta\end{smallmatrix}\right)$，
  其方向与复数乘以 $e^{i\theta}$ 不一致。
  上文改为列向量左乘该矩阵。
\end{enumerate}

% 第 16 张板书结束；第 17 张继续讨论倒数映射与圆。


% 第 17 张板书：倒数变换下圆、直线方程的变化。
\paragraph{倒数变换与圆、直线}
圆的一般方程可写为
\[
  x^2+y^2+Ax+By+C=0,
\]
亦即
\[
  z\overline z+\overline\alpha z+\alpha\overline z+\beta=0,
  \qquad
  \alpha=\frac{A+iB}{2},\quad \beta=C\in\mathbb R.
\]
令
\[
  w=\frac1z,\qquad z=\frac1w,
\]
则
\begin{align*}
  &\frac{1}{w\overline w}
   +\frac{\overline\alpha}{w}
   +\frac{\alpha}{\overline w}+\beta=0\\
  \iff{}&
  \beta w\overline w+\alpha w+\overline\alpha\,\overline w+1=0.
\end{align*}
分两种情况：
\[
  \begin{cases}
    \beta=0:\quad
      \alpha w+\overline\alpha\,\overline w+1=0,
      &\text{直线},\\[2pt]
    \beta\ne0:\quad
      w\overline w+\dfrac{\alpha}{\beta}w
      +\dfrac{\overline\alpha}{\beta}\overline w
      +\dfrac1{\beta}=0,
      &\text{圆}.
  \end{cases}
\]
这里的圆均指非退化圆。因而倒数变换将圆映为圆或直线；
其中过原点的圆（$\beta=0$）映为直线。

% 第 18 张板书：基本变换与一般 Möbius 变换的分解。
\paragraph{结论：分式线性映射保持广义圆}
广义圆指圆与直线。考虑三种基本变换：
\[
  \text{① 平移：}\quad w=z+\alpha;\qquad
  \text{② 伸缩旋转：}\quad w=\alpha z\ (\alpha\ne0);\qquad
  \text{③ 倒数：}\quad w=\frac1z.
\]
这些变换均保持广义圆的类型（圆或直线的集合）。

对于一般 Möbius 变换
\[
  w=\frac{az+b}{cz+d},\qquad ad-bc\ne0,
\]
若 $c\ne0$，按板书方法整理：
\begin{align*}
  w
  &=\frac{a\left(z+\dfrac dc\right)+
             \left(b-\dfrac{ad}{c}\right)}
           {c\left(z+\dfrac dc\right)}\\
  &=\frac ac+\frac{bc-ad}{c^2}\,
        \frac1{z+\dfrac dc}.
\end{align*}
记
\[
  \alpha'=\frac{bc-ad}{c^2}\ne0,
\]
于是
\[
  \boxed{w=\frac ac+\frac{\alpha'}{z+d/c}}.
\]
因此可按以下顺序由基本变换复合得到：
\[
  z
  \xmapsto{\text{①}}z+\frac dc
  \xmapsto{\text{③}}\frac1{z+d/c}
  \xmapsto{\text{②}}\frac{\alpha'}{z+d/c}
  \xmapsto{\text{①}}\frac ac+\frac{\alpha'}{z+d/c}.
\]
即板书中的 \textbf{① $\to$ ③ $\to$ ② $\to$ ①}。

若 $c=0$，由于 $ad-bc=ad\ne0$，
\[
  w=\frac ad\,z+\frac bd,
\]
只需伸缩旋转与平移两步。因此一般分式线性映射
将广义圆映为广义圆。

\paragraph{板书勘误（第 18 张）}
在引入 $z+t$ 的中间变形中，需同时减去所增加的项：
\[
  az+b=a(z+t)+(b-at).
\]
若直接写作 $a(z+t)+b$，则不与原分子相等；
上文已用 $t=d/c$ 补齐该项。板书最终的分解系数
$\alpha'=(bc-ad)/c^2$ 是正确的。
