\section{留数（Residues）}

\begin{proposition}[解析函数零点的孤立性]
若 $f(z)$ 在 $z_0$ 点解析，$f(z_0)=0$，且 $f(z)\not\equiv 0$，
则存在 $\delta>0$，使得在 $|z-z_0|<\delta$ 内有
\[
  f(z)=(z-z_0)^n\varphi(z),
\]
这里 $n\in\mathbb{N}^{+}$，$\varphi(z_0)\ne 0$，且
$\varphi(z)$ 在 $|z-z_0|<\delta$ 内处处可导（解析）。
\end{proposition}

\paragraph{证明}
因 $f(z)$ 在 $z_0$ 解析，故存在 $\delta>0$，当 $|z-z_0|<\delta$ 时，有
\begin{align*}
  f(z)
  &=\sum_{k=0}^{+\infty}\frac{f^{(k)}(z_0)}{k!}(z-z_0)^k \\
  &=f(z_0)+\frac{f'(z_0)}{1!}(z-z_0)+\cdots
    +\frac{f^{(n)}(z_0)}{n!}(z-z_0)^n+\cdots.
\end{align*}
令
\[
  n=\min\bigl\{k: f^{(k)}(z_0)\ne 0\bigr\},
\]
则
\[
  f(z_0)=f'(z_0)=\cdots=f^{(n-1)}(z_0)=0.
\]
于是
\begin{align*}
  f(z)
  &=\frac{f^{(n)}(z_0)}{n!}(z-z_0)^n
    +\frac{f^{(n+1)}(z_0)}{(n+1)!}(z-z_0)^{n+1}+\cdots \\
  &=(z-z_0)^n\underbrace{\left[
      \frac{f^{(n)}(z_0)}{n!}
      +\frac{f^{(n+1)}(z_0)}{(n+1)!}(z-z_0)+\cdots
    \right]}_{\displaystyle\varphi(z)} \\
  &=(z-z_0)^n\varphi(z).
\end{align*}
% 2026-10-19，第 2 张板书：续证命题 1、命题 2。
由于
\[
  \varphi(z)
  =\frac{f^{(n)}(z_0)}{n!}
    +\frac{f^{(n+1)}(z_0)}{(n+1)!}(z-z_0)+\cdots,
  \qquad
  \varphi(z_0)=\frac{f^{(n)}(z_0)}{n!}\ne 0.
\]
又因为 $\varphi(z)$ 在 $z_0$ 解析，所以在 $z_0$ 连续。
由连续函数在一点不为零的性质可知，
存在 $\delta>0$，使得
\[
  \varphi(z)\ne 0,\qquad
  \forall z\in\{z:|z-z_0|<\delta\}.
\]
因此，在该邻域内，由 $f(z)=(z-z_0)^n\varphi(z)$ 可知，
$f(z)=0$ 当且仅当 $z=z_0$。
即 $z_0$ 是 $f(z)$ 在邻域 $|z-z_0|<\delta$ 内的唯一零点。
\hfill$\square$

\begin{proposition}[解析函数唯一性定理]
设 $D$ 为区域，$f(z)$、$g(z)$ 在 $D$ 内解析。
若存在 $D$ 中一列互异点 $z_1,z_2,\ldots,z_n,\ldots$，满足
\[
  z_n\longrightarrow z^*\in D,
  \qquad f(z_n)=g(z_n)\quad(n=1,2,\ldots),
\]
则
\[
  f(z)\equiv g(z),\qquad \forall z\in D.
\]
\end{proposition}

\paragraph{证明}
令
\[
  h(z)=f(z)-g(z).
\]
则 $h(z)$ 在 $D$ 内解析，且
\[
  h(z_1)=h(z_2)=\cdots=h(z_n)=\cdots=0.
\]
由 $z_n\to z^*\in D$ 及 $h(z)$ 在 $z^*$ 处连续，知
\[
  h(z^*)=\lim_{n\to\infty}h(z_n)=0.
\]
若 $f(z)\not\equiv g(z)$，则 $h(z)\not\equiv0$。
根据命题~1（解析函数零点的孤立性），$z^*$ 必为
$h(z)$ 的孤立零点；即存在 $\delta>0$，
在 $0<|z-z^*|<\delta$ 内 $h(z)\ne0$。
但互异的零点列 $z_n\to z^*$，必有某个 $z_n$ 满足
\[
  0<|z_n-z^*|<\delta,\qquad h(z_n)=0,
\]
矛盾。因此 $h(z)\equiv0$，即
\[
  \boxed{f(z)\equiv g(z)\quad(z\in D).}
\]
\hfill$\square$


% 2026-10-19，第 3 张板书：奇点的分类。
\subsection*{奇点的分类（Singularities）}

奇点分为\textbf{孤立奇点}与\textbf{非孤立奇点}。
孤立奇点进一步分为：
\[
  \left\{
  \begin{array}{l}
    \text{极点（pole）},\\
    \text{可去奇点（removable singularity）},\\
    \text{本性奇点（essential singularity）}.
  \end{array}
  \right.
\]

\paragraph{非孤立奇点}
例如
\[
  f(z)=\frac{1}{\sin\frac{1}{z}},\qquad z\ne 0.
\]
取
\[
  z_k=\frac{1}{k\pi},\qquad k\in\mathbb Z\setminus\{0\},
\]
则 $z_k$ 是奇点，且 $z_k\to0$（当 $|k|\to\infty$）。
所以 $z_0=0$ 是一个\textbf{非孤立奇点}。

另一个板书例子是
\[
  f(z)=|z|^2=z\overline z.
\]
它在 $z_0=0$ 处复可导，且 $f'(0)=0$；
但在任何包含 $0$ 的邻域内均不解析。
% 注：在一点复可导并不意味着该点解析。

\paragraph{极点（pole）}
设 $z_0$ 为 $f(z)$ 的孤立奇点。
若 $f(z)$ 在 $z_0$ 附近的 Laurent 级数为
\begin{align*}
  f(z)
  &=\frac{C_{-n}}{(z-z_0)^n}
   +\frac{C_{-(n-1)}}{(z-z_0)^{n-1}}
   +\cdots\\
  &\quad +C_0+C_1(z-z_0)+\cdots+C_n(z-z_0)^n+\cdots,
\end{align*}
其中 $C_{-n}\ne0$，$n\in\mathbb N^+$，
则称 $z_0$ 为 $f(z)$ 的\textbf{$n$ 阶极点}。
此时
\[
  \lim_{z\to z_0}f(z)=\infty
  \qquad\bigl(\text{即 }|f(z)|\to\infty\bigr).
\]

\paragraph{可去奇点（removable singularity）}
设 $z_0$ 为 $f(z)$ 的孤立奇点。
若其 Laurent 级数的所有负幂项系数为零，
即 $C_{-n}=0$（$\forall n\ge1$），则
\[
  f(z)=C_0+C_1(z-z_0)+C_2(z-z_0)^2+\cdots,
\]
即在去心邻域中表现为 Taylor 级数，并且
\[
  \lim_{z\to z_0}f(z)=C_0.
\]
在 $z_0$ 处补充定义 $f(z_0)=C_0$，即可使其在 $z_0$ 处解析。


% 2026-10-19，第 4 张板书：本性奇点及零点分解式的补充。
\paragraph{本性奇点（essential singularity）}
若 $z_0$ 为 $f(z)$ 的孤立奇点，且 $f(z)$ 在
$z_0$ 的去心邻域内的 Laurent 展开
\[
  f(z)=\cdots+\frac{C_{-n}}{(z-z_0)^n}
        +\cdots+C_0+C_1(z-z_0)+\cdots
\]
含有\textbf{无穷多个非零的负幂项系数}，即存在无穷多个
$n\in\mathbb N^+$ 使得 $C_{-n}\ne0$，则称 $z_0$
为 $f(z)$ 的本性奇点。

\paragraph{例}
考虑
\[
  f(z)=e^{1/z},\qquad z_0=0.
\]
由于
\[
  e^{1/z}
  =\sum_{n=0}^{+\infty}\frac{1}{n!\,z^n}
  =\sum_{n=0}^{+\infty}\frac{z^{-n}}{n!},
\]
其 Laurent 级数有无穷多个非零的负幂项系数
$C_{-n}=1/n!$，所以 $z=0$ 是本性奇点。

此时 $\lim_{z\to0}f(z)$ 不存在。例如沿虚轴取
$z=i y_n$，其中 $y_n\in\mathbb R\setminus\{0\}$
且 $y_n\to0$，则
\[
  f(iy_n)=e^{1/(iy_n)}=e^{-i/y_n},
  \qquad |f(iy_n)|=1.
\]
另一方面，沿正实轴 $x\to0^+$ 时
$e^{1/x}\to+\infty$，因此在 $z=0$ 不存在极限
（也不趋于无穷大）。

\paragraph{补充：零点分解式中的可去奇点}
回顾前述解析函数在零点 $z_0$ 附近的分解
\[
  f(z)=(z-z_0)^n\varphi(z).
\]
在 $z\ne z_0$ 时，可由 Taylor 展开得到
\begin{align*}
  \varphi(z)
   &=\frac{f(z)}{(z-z_0)^n}\\
   &=\frac{
      \dfrac{f^{(n)}(z_0)}{n!}(z-z_0)^n+
      \dfrac{f^{(n+1)}(z_0)}{(n+1)!}(z-z_0)^{n+1}
      +\cdots}{(z-z_0)^n}\\
   &=\frac{f^{(n)}(z_0)}{n!}
     +\frac{f^{(n+1)}(z_0)}{(n+1)!}(z-z_0)
     +O\bigl((z-z_0)^2\bigr).
\end{align*}
因此
\[
  \lim_{z\to z_0}\varphi(z)
  =\frac{f^{(n)}(z_0)}{n!}\ne0,
\]
该极限存在且有限。把
$\varphi(z_0)=f^{(n)}(z_0)/n!$ 补充定义在 $z_0$
处，即可将 $\varphi$ 在 $z_0$ 的可去奇点补为解析点；
其 Laurent 级数因此化为 Taylor 级数。
% 注意：这里的可去奇点属于最初只在 z != z_0 定义的商函数
% \varphi(z)=f(z)/(z-z_0)^n，而非已在 z_0 解析的 f。


% 2026-10-19，第 5 张板书：三角恒等式及复函数的洛必达法则。
\subsection*{唯一性定理的应用：三角恒等式与欧拉公式}

\paragraph{例 1：复数范围的三角恒等式}
设
\[
  F(z)=\sin^2 z+\cos^2 z.
\]
则 $F(z)$ 是整函数，且
\[
  F'(z)=2\sin z\cos z-2\cos z\sin z\equiv0.
\]
因而 $F(z)$ 为常数。由 $F(0)=1$，得到
\[
  \boxed{\sin^2 z+\cos^2 z\equiv1,\quad z\in\mathbb C.}
\]
也可以从实数范围内的恒等式
$\sin^2\theta+\cos^2\theta=1$（$\theta\in\mathbb R$）出发，
利用解析函数唯一性定理将其推广到整个复平面。

\paragraph{例 2：欧拉公式的解析延拓}
对任意 $\theta\in\mathbb R$，已知
\[
  e^{i\theta}=\cos\theta+i\sin\theta.
\]
由于 $e^z$、$\cos z$ 和 $\sin z$ 都是整函数
（特别地 $(e^z)'=e^z$），考察
\[
  H(z)=e^{iz}-\cos z-i\sin z.
\]
对所有 $\theta\in\mathbb R$ 都有 $H(\theta)=0$。
由解析函数唯一性定理，得到
\[
  H(z)\equiv0,\qquad
  \boxed{e^{iz}=\cos z+i\sin z,\quad z\in\mathbb C.}
\]

\begin{proposition}[复函数的洛必达法则（L'Hospital 法则）]
若 $f(z)$、$g(z)$ 在 $z_0$ 解析，
且 $f(z_0)=g(z_0)=0$；
在 $z_0$ 的某个去心邻域中 $f(z)\ne0$、$g(z)\ne0$，
则在扩充复平面的意义下，
\[
  \boxed{\lim_{z\to z_0}\frac{f(z)}{g(z)}
   =\lim_{z\to z_0}\frac{f'(z)}{g'(z)}.}
\]
上述两个极限均可为有限值或 $\infty$；
导数商在足够小的去心邻域内有定义。
\end{proposition}

\paragraph{证明}
由解析函数零点的孤立性定理，存在 $n,m\in\mathbb N^+$
以及在 $z_0$ 邻域内解析的函数 $\varphi(z),\psi(z)$，
使得
\[
  \begin{cases}
    f(z)=(z-z_0)^n\varphi(z),\\
    g(z)=(z-z_0)^m\psi(z),
  \end{cases}
  \qquad
  \varphi(z_0)\ne0,\quad \psi(z_0)\ne0.
\]
因此
\[
  \frac{f(z)}{g(z)}
  =(z-z_0)^{n-m}\frac{\varphi(z)}{\psi(z)}.
\]
由 $\varphi(z_0),\psi(z_0)\ne0$ 可得
\[
  \lim_{z\to z_0}\frac{f(z)}{g(z)}
  =
  \begin{cases}
     0, & n>m,\\[2pt]
     \displaystyle\frac{\varphi(z_0)}{\psi(z_0)}
       \ne0,\infty, & n=m,\\[7pt]
     \infty, & n<m.
  \end{cases}
\]

另一方面，求导可得
\[
  \begin{aligned}
    f'(z)&=n(z-z_0)^{n-1}\varphi(z)
             +(z-z_0)^n\varphi'(z),\\
    g'(z)&=m(z-z_0)^{m-1}\psi(z)
             +(z-z_0)^m\psi'(z).
  \end{aligned}
\]
所以
\[
  \frac{f'(z)}{g'(z)}
  =\frac{n(z-z_0)^{n-1}\varphi(z)+(z-z_0)^n\varphi'(z)}
         {m(z-z_0)^{m-1}\psi(z)+(z-z_0)^m\psi'(z)}
  =(z-z_0)^{n-m}
  \frac{n\varphi(z)+(z-z_0)\varphi'(z)}
       {m\psi(z)+(z-z_0)\psi'(z)}.
\]
于是
\[
  \lim_{z\to z_0}\frac{f'(z)}{g'(z)}
  =
  \begin{cases}
     0, & n>m,\\[2pt]
     \displaystyle\frac{n\varphi(z_0)}{m\psi(z_0)}
      =\frac{\varphi(z_0)}{\psi(z_0)}, & n=m,\\[7pt]
     \infty, & n<m.
  \end{cases}
\]
逐项比较两式，得到
\[
  \boxed{\lim_{z\to z_0}\frac{f(z)}{g(z)}
  =\lim_{z\to z_0}\frac{f'(z)}{g'(z)}.}
\]
\hfill$\square$


% 2026-10-19，第 6、7 张板书：实变量中值定理的局限、C_{-1} 的计算及例题。
\subsection*{补充：复变洛必达法则为何不能照搬实变中值定理}

\paragraph{板书中的想法}
对\emph{实变量} $x$，假设 $f,g$ 在实区间满足拉格朗日中值定理的条件，
且 $f(x_0)=g(x_0)=0$，则存在夹在 $x_0$ 与 $x$ 之间的
$\xi_x,\eta_x$，使得
\begin{align*}
  f(x)-f(x_0)&=f'(\xi_x)(x-x_0),\\
  g(x)-g(x_0)&=g'(\eta_x)(x-x_0).
\end{align*}
于是（在分母非零时）
\[
  \frac{f(x)-f(x_0)}{g(x)-g(x_0)}
  =\frac{f'(\xi_x)(x-x_0)}
         {g'(\eta_x)(x-x_0)}
  =\frac{f'(\xi_x)}{g'(\eta_x)}.
\]
当 $x\to x_0$ 时，区间内的中间点满足
$\xi_x\to x_0$、$\eta_x\to x_0$。若另有
$f',g'$ 在 $x_0$ 连续、$g'(x_0)\ne0$，则可推出
\[
  \lim_{x\to x_0}
    \frac{f(x)-f(x_0)}{g(x)-g(x_0)}
  =\frac{f'(x_0)}{g'(x_0)}.
\]

\paragraph{为什么老师在板书中将这一步划掉？（补全说明）}
上述是实数区间上的推导，\textbf{不能直接推广到复变量 $z$}：
复值解析函数一般不满足形如
$f(z)-f(z_0)=f'(\xi)(z-z_0)$ 的拉格朗日中值定理，
因此不能在复平面中凭空引入上述中间点 $\xi,\eta$。
例如 $f(z)=e^z$，取 $z_0=0$、$z=2\pi i$，
则 $f(z)-f(z_0)=0$，但 $f'(\xi)=e^\xi\ne0$
对所有 $\xi\in\mathbb C$ 均成立。
这就排除了相应的复数版中值定理。

同时，实变量推导中出现的 $\xi_x$、$\eta_x$
是\emph{两个不同的中间点}；
在一般的实变洛必达定理中，不能未经额外条件
便将 $f'(\xi_x)/g'(\eta_x)$
视为相同点处的导数比 $f'(t)/g'(t)$。
实变法则通常使用柯西中值定理；
\textbf{复变情形则使用之前板书中的零点分解式（Taylor 展开）证明}，
无需依赖中值定理。

\subsection*{Laurent 系数 $C_{-1}$ 的计算}

\paragraph{一阶极点的系数}
设 $z_0$ 是 $f(z)$ 的一阶极点，其 Laurent 展开为
\[
  f(z)=\frac{C_{-1}}{z-z_0}
       +C_0+C_1(z-z_0)+\cdots.
\]
两侧乘以 $z-z_0$，有
\[
  (z-z_0)f(z)=C_{-1}+C_0(z-z_0)
     +C_1(z-z_0)^2+\cdots.
\]
令 $z\to z_0$，可得
\[
  \boxed{C_{-1}=\lim_{z\to z_0}(z-z_0)f(z).}
\]
这里 $C_{-1}$ 是 $(z-z_0)^{-1}$ 项的系数，也称
$f$ 在 $z_0$ 处的\textbf{留数}（residue）。

\paragraph{常用公式：$f=p/g$ 且分母有一阶零点}
若
\[
  f(z)=\frac{p(z)}{g(z)},\qquad
  p(z_0)\ne0,\quad g(z_0)=0,\quad g'(z_0)\ne0,
\]
且 $p,g$ 在 $z_0$ 附近解析，
则 $z_0$ 是 $f$ 的一阶极点。
由于
\[
  g(z)=g'(z_0)(z-z_0)+o(z-z_0),
  \qquad
  p(z)=p(z_0)+p'(z_0)(z-z_0)+o(z-z_0),
\]
因此
\begin{align*}
  C_{-1}
  &=\lim_{z\to z_0}(z-z_0)\frac{p(z)}{g(z)}\\
  &=\frac{p(z_0)}{g'(z_0)}.
\end{align*}
故
\[
  \boxed{\operatorname{Res}(f,z_0)
    =C_{-1}=\frac{p(z_0)}{g'(z_0)}.}
\]

\paragraph{例 1：$\boldsymbol{\sin(z^2)/z^3}$ 在原点的留数}
板书中的展开式表明，本例取
\[
  f(z)=\frac{\sin(z^2)}{z^3},\qquad z_0=0.
\]
利用 $\sin w=w-w^3/3!+w^5/5!-\cdots$，有
\begin{align*}
  f(z)
  &=\frac{z^2-\dfrac{z^6}{3!}
                +\dfrac{z^{10}}{5!}-\cdots}{z^3}\\
  &=\frac1z-\frac{z^3}{3!}+\frac{z^7}{5!}-\cdots.
\end{align*}
所以 $z=0$ 是一阶极点，且
\[
  \boxed{C_{-1}
  =\lim_{z\to0}z f(z)
  =\lim_{z\to0}\frac{\sin(z^2)}{z^2}
  =1.}
\]
\emph{注意：}不能直接将本例代入上面的
$p(z_0)/g'(z_0)$ 公式，因为若取
$p(z)=\sin(z^2)$、$g(z)=z^3$，
则 $p(0)=g'(0)=0$。应先约去公因子或直接展开。

\paragraph{例 2：$\boldsymbol{1/(1+z^n)}$ 在各极点的留数}
设 $n\in\mathbb N^+$，取
\[
  f(z)=\frac{1}{1+z^n}.
\]
分母零点满足 $z^n=-1$。
将板书中含糊的 $(-1)^{1/n}$ 明确写为 $n$ 个不同的根：
\[
  \boxed{z_k=e^{\,i(2k+1)\pi/n},
    \qquad k=0,1,\ldots,n-1.}
\]
取 $p(z)=1$、$g(z)=1+z^n$。则
\[
  g(z_k)=0,\qquad g'(z_k)=nz_k^{n-1}\ne0,
\]
故每个 $z_k$ 都是一阶极点，其 Laurent 系数为
\begin{align*}
  C_{-1}^{(k)}
  &=\operatorname{Res}(f,z_k)
   =\frac{p(z_k)}{g'(z_k)}\\
  &=\frac{1}{n z_k^{n-1}}
   =-\frac{z_k}{n}.
\end{align*}
其中最后一步用了 $z_k^n=-1$。


% 板书第 8 张：高阶极点留数系数 C_{-1} 的计算。
\paragraph{$n$ 阶极点处 $C_{-1}$ 的导数公式}
设 $z_0$ 为 $f(z)$ 的 $n$ 阶极点（$n\in\mathbb N^+$），
则其 Laurent 展开为
\[
  f(z)=\frac{C_{-n}}{(z-z_0)^n}
  +\frac{C_{-(n-1)}}{(z-z_0)^{n-1}}
  +\cdots+\frac{C_{-1}}{z-z_0}
  +C_0+C_1(z-z_0)+\cdots,
  \qquad C_{-n}\ne0.
\]
两边同乘 $(z-z_0)^n$，得到
\begin{align*}
  h(z):=(z-z_0)^n f(z)
   &=C_{-n}+C_{-(n-1)}(z-z_0)+\cdots\\
   &\quad+C_{-1}(z-z_0)^{n-1}
   +C_0(z-z_0)^n+\cdots.
\end{align*}
乘积 $h(z)$ 在 $z_0$ 处具有可去奇点，将其补为
$h(z_0)=C_{-n}$ 后，$h$ 在 $z_0$ 附近解析。

对上式两侧求 $(n-1)$ 阶导数，再令 $z=z_0$：
低于 $(n-1)$ 次的多项式项求导后为零；
高于 $(n-1)$ 次的项在 $z=z_0$ 处为零。
所以恰好只剩
\[
  h^{(n-1)}(z_0)=(n-1)!\,C_{-1}.
\]
因此
\[
  \boxed{
    C_{-1}=\operatorname{Res}(f,z_0)
    =\frac{1}{(n-1)!}
       \lim_{z\to z_0}
       \frac{d^{\,n-1}}{dz^{\,n-1}}
       \bigl[(z-z_0)^n f(z)\bigr]
  }.
\]
当 $n=1$ 时退化为之前的一阶极点公式
\[
  C_{-1}=\lim_{z\to z_0}(z-z_0)f(z).
\]
若 $f(z)=p(z)/g(z)$ 且
$p(z_0)\ne0$、$g(z_0)=0$、$g'(z_0)\ne0$，
则有
\[
  C_{-1}=\frac{p(z_0)}{g'(z_0)}.
\]

% 板书第 9 张：例题：1/(1+z^n) 在大圆周上的积分。
\subsection*{围道积分例题：$\displaystyle\oint\frac{dz}{1+z^n}$}
\paragraph{题目}
设 $n\in\mathbb N^+$，$r>1$，圆周
$\Gamma_r:|z|=r$ 取正向（逆时针方向）。求
\[
  I_{r,n}=\oint_{\Gamma_r}\frac{1}{1+z^n}\,dz.
\]
由于 $1+z^n$ 的零点均在 $|z|=1$ 上，
故当 $r>1$ 时，积分路径不经过任何极点。

\paragraph{解法一：变量代换 $z=1/t$}
令
\[
  t=\frac1z,\qquad dz=-\frac{dt}{t^2}.
\]
当 $z$ 沿 $\Gamma_r$ 逆时针行进时，
$t$ 沿 $|t|=1/r<1$ 顺时针行进。
记顺时针曲线为 $\Gamma_{1/r}^{-}$，则
\begin{align*}
  I_{r,n}
  &=\oint_{\Gamma_{1/r}^{-}}
       \frac{1}{1+t^{-n}}\left(-\frac{dt}{t^2}\right)\\
  &=\oint_{|t|=1/r}^{\mathrm{CCW}}
       \frac{t^{n-2}}{1+t^n}\,dt.
\end{align*}
这里的两个负号分别来自 $dz=-dt/t^2$
以及积分方向的反转，故相互抵消。

当 $n=1$ 时，
\[
  \frac{t^{n-2}}{1+t^n}=\frac{1}{t(1+t)},
\]
在 $|t|<1$ 内仅有一阶极点 $t=0$，
留数为 $1$，因此
\[
  I_{r,1}=2\pi i.
\]
当 $n\ge2$ 时，$t^{n-2}/(1+t^n)$
在闭圆盘 $|t|\le1/r<1$ 上解析。
由柯西积分定理，$I_{r,n}=0$。
所以
\[
  \boxed{I_{r,n}=
    \begin{cases}
      2\pi i, & n=1,\\
      0, & n\ge2.
    \end{cases}}
\]

\paragraph{解法二：留数定理（与上一例衔接）}
令
\[
  f(z)=\frac{p(z)}{g(z)}=\frac{1}{1+z^n},
  \qquad p(z)=1,\quad g(z)=1+z^n.
\]
所有 $n$ 个极点为
\[
  z_k=e^{i(2k-1)\pi/n},\qquad k=1,\ldots,n,
  \qquad |z_k|=1<r,
\]
且
\[
  g'(z_k)=nz_k^{n-1}\ne0.
\]
所以每个 $z_k$ 都是一阶极点，
对应的 Laurent 系数（留数）为
\[
  C_{-1}^{(k)}
    =\frac{p(z_k)}{g'(z_k)}
    =\frac{1}{nz_k^{n-1}}
    =-\frac{z_k}{n}.
\]
由留数定理，
\begin{align*}
  I_{r,n}
   &=2\pi i\sum_{k=1}^{n}C_{-1}^{(k)}\\
   &=\frac{2\pi i}{n}
      \sum_{k=1}^{n}\frac{1}{z_k^{n-1}}
    =-\frac{2\pi i}{n}\sum_{k=1}^{n}z_k.
\end{align*}
若 $n=1$，唯一的根是 $z_1=-1$，故
$I_{r,1}=2\pi i$。

若 $n\ge2$，这些根构成等比数列：
\[
  \sum_{k=1}^{n}z_k
  =e^{i\pi/n}\sum_{k=0}^{n-1}e^{2\pi ik/n}
  =e^{i\pi/n}
      \frac{1-e^{2\pi i}}{1-e^{2\pi i/n}}
  =0.
\]
因此 $I_{r,n}=0$，与解法一一致。


% 第 10 张板书：根的求和与参数化圆周积分。
\subsection*{前一例补充：利用多项式系数计算根的和}
沿用前一题 $z_k^n=-1$（$k=1,\ldots,n$）的记号。
因为 $z_k^n=-1$，故
\[
   \frac1{z_k^{n-1}}=-z_k,
   \qquad
   I_{r,n}=-\frac{2\pi i}{n}\sum_{k=1}^{n}z_k.
\]
另一方面，按这些互异根分解多项式，
\[
  z^n+1=\prod_{k=1}^{n}(z-z_k)
  =z^n-\left(\sum_{k=1}^{n}z_k\right)z^{n-1}
  +\cdots+(-1)^n\prod_{k=1}^{n}z_k.
\]
当 $n\ge2$ 时，左侧的 $z^{n-1}$ 项系数为 $0$，
故根据韦达定理
\[
  \sum_{k=1}^{n}z_k=0,\qquad I_{r,n}=0\quad(n\ge2).
\]
当 $n=1$ 时，唯一的根为 $z_1=-1$，所以
$I_{r,1}=2\pi i$。这给出了上一题结果的另一种解释。

\subsection*{含参数 $r$ 的圆周积分}
设 $r>0$，$n\in\mathbb N^+$，$R>r$。
以下所有圆周 $\Gamma_R:|z|=R$ 均取逆时针正向。
为避免与上一例的记号混淆，将两个新积分分别记为
\[
  A_{R,r,n}:=\oint_{\Gamma_R}\frac{dz}{r^n+z^n},
  \qquad
  B_{R,r,n}:=\oint_{\Gamma_R}\frac{z^{2n}}{r^n+z^n}\,dz.
\]

\paragraph{第一题：$\boldsymbol{A_{R,r,n}}$，利用倒数变换}
令 $z=1/t$，则 $dz=-dt/t^2$。
当 $z$ 沿 $|z|=R$ 逆时针行进时，
$t$ 沿 $|t|=1/R$ 顺时针行进。于是
\begin{align*}
  A_{R,r,n}
  &=\oint_{\substack{|t|=1/R\\\mathrm{clockwise}}}
      \frac{1}{r^n+t^{-n}}
      \left(-\frac{dt}{t^2}\right)\\
  &=\oint_{\substack{|t|=1/R\\\mathrm{counterclockwise}}}
      \frac{t^{n-2}}{1+r^n t^n}\,dt.
\end{align*}
这里负号和曲线方向反转正好抵消。
由于 $1/R<1/r$，分母 $1+r^nt^n$ 在
$|t|\le 1/R$ 内没有零点。

若 $n=1$，被积函数为
$\frac{1}{t(1+rt)}$，
圆内只有 $t=0$ 这一一阶极点，其留数为 $1$，
故 $A_{R,r,1}=2\pi i$。

若 $n\ge2$，$t^{n-2}/(1+r^nt^n)$
在 $|t|\le1/R$ 内解析，
由柯西积分定理 $A_{R,r,n}=0$。
因此
\[
  \boxed{
    A_{R,r,n}
    =\oint_{|z|=R}\frac{dz}{r^n+z^n}
    =\begin{cases}
       2\pi i,& n=1,\\
       0,& n\ge2.
      \end{cases}
    \qquad R>r>0.
  }
\]

\paragraph{第二题：$\boldsymbol{B_{R,r,n}}$，先作代数分解}
由平方差公式，
\[
  z^{2n}
  =z^{2n}-r^{2n}+r^{2n}
  =(z^n-r^n)(z^n+r^n)+r^{2n}.
\]
所以
\[
  \frac{z^{2n}}{r^n+z^n}
  =z^n-r^n+\frac{r^{2n}}{r^n+z^n}.
\]
沿圆周积分，
\begin{align*}
  B_{R,r,n}
  &=\oint_{\Gamma_R}(z^n-r^n)\,dz
    +r^{2n}\oint_{\Gamma_R}\frac{dz}{r^n+z^n}\\
  &=0+r^{2n}A_{R,r,n}.
\end{align*}
第一项为零，因为 $z^n-r^n$ 是整函数。
代入第一题的结果，
\[
  \boxed{
    B_{R,r,n}
    =\oint_{|z|=R}\frac{z^{2n}}{r^n+z^n}\,dz
    =\begin{cases}
       2\pi i\,r^2,&n=1,\\
       0,&n\ge2.
      \end{cases}
    \qquad R>r>0.
  }
\]


% 第 11 张板书：考试重点——含 cos/sin 分母的三角积分。
\subsection*{考试重点：三角函数分母的定积分（必会例题）}

\begin{center}
\fbox{\begin{minipage}{0.94\linewidth}
\centering
\textbf{【考试重点｜老师明确强调会考】}\\[4pt]
\textbf{题目：}设 $a,b\in\mathbb R$，$a>|b|\ge0$，计算
\[
  I_{a,b}=\int_0^{2\pi}\frac{d\theta}{a+b\cos\theta}
         =\int_0^{2\pi}\frac{d\theta}{a+b\sin\theta}.
\]
\textbf{标准答案：}
\[
  \boxed{\displaystyle I_{a,b}=\frac{2\pi}{\sqrt{a^2-b^2}}}.
\]
\end{minipage}}
\end{center}

\paragraph{第一步：解释老师板书中的余弦--正弦积分等式}
对连续函数 $F\in C^0([-1,1])$，将
$\theta\mapsto\theta+\pi/2$ 并利用 $2\pi$ 周期性，得
\begin{align*}
  \int_0^{2\pi} F(\cos\theta)\,d\theta
  &=\int_{\pi/2}^{5\pi/2}F(\sin t)\,dt\\
  &=\int_0^{2\pi}F(\sin t)\,dt.
\end{align*}
其中 $\cos\theta=\sin(\theta+\pi/2)$。
而老师同时在板书中写了第一象限的等式
\[
  \int_0^{\pi/2}F(\cos\theta)\,d\theta
  =\int_0^{\pi/2}F(\sin\theta)\,d\theta,
\]
它可由 $t=\pi/2-\theta$ 得到；对其他三个象限
也可分别作 $t=\pi-\theta$、$t=3\pi/2-\theta$、
$t=2\pi-\theta$ 等代换，再比较对应区间的积分。
对本题取 $F(x)=1/(a+bx)$ 即可，因为
$a>|b|$ 保证 $a+bx>0$（$-1\le x\le1$）。

\paragraph{第二步：留数定理法（复变函数考试标准解答）}
令 $z=e^{i\theta}$，当 $\theta$ 从 $0$ 增至 $2\pi$ 时，
$z$ 沿单位圆 $|z|=1$ 逆时针一周，且
\[
  \cos\theta=\frac12\left(z+\frac1z\right),
  \qquad d\theta=\frac{dz}{iz}.
\]
因此
\begin{align*}
  I_{a,b}
  &=\oint_{|z|=1}
      \frac{1}{a+\frac b2(z+z^{-1})}\frac{dz}{iz}\\
  &=\frac{2}{i}\oint_{|z|=1}
      \frac{dz}{bz^2+2az+b}.
\end{align*}
记 $\Delta=\sqrt{a^2-b^2}>0$。
若 $b\ne0$，分母的两个根为
\[
  z_{\pm}=\frac{-a\pm\Delta}{b}.
\]
其中
\[
  z_+=\frac{-a+\Delta}{b}
      =-\frac{b}{a+\Delta},
  \qquad |z_+|=\frac{|b|}{a+\Delta}<1,
\]
而由 $z_+z_-=1$ 知 $|z_-|>1$。
因此单位圆内部只有一个一阶极点 $z_+$。
在这个极点处，
\[
  \operatorname{Res}_{z=z_+}
   \frac{2/i}{bz^2+2az+b}
  =\frac{2/i}{2bz_++2a}
  =\frac{1}{i\Delta},
\]
其中使用了 $bz_++a=\Delta$。
由留数定理，
\[
  I_{a,b}
   =2\pi i\cdot\frac{1}{i\Delta}
   =\boxed{\frac{2\pi}{\sqrt{a^2-b^2}}}.
\]
\textbf{特例 $b=0$：} 原积分直接为
$\int_0^{2\pi}d\theta/a=2\pi/a$；
由于 $a>0$，它仍符合上面的统一公式。

\paragraph{附：不用复变函数的快速手算解法}
若题目只要求计算而不指定使用留数定理，
可使用半角正切代换 $t=\tan(\theta/2)$。
注意 $\theta=\pi$ 时 $t$ 经过无穷远点：
必须分别处理 $[0,\pi)$ 与 $(\pi,2\pi]$ 两段。
由
\[
  \cos\theta=\frac{1-t^2}{1+t^2},
  \qquad d\theta=\frac{2\,dt}{1+t^2},
\]
得到
\begin{align*}
  I_{a,b}
  &=2\int_{-\infty}^{+\infty}
       \frac{dt}{(a+b)+(a-b)t^2}\\
  &=\frac{2}{\sqrt{(a+b)(a-b)}}
       \left[
        \arctan\left(t\sqrt{\frac{a-b}{a+b}}\right)
       \right]_{-\infty}^{+\infty}\\
  &=\frac{2\pi}{\sqrt{a^2-b^2}}.
\end{align*}
因 $a>|b|$，有 $a+b>0$、$a-b>0$，
所以代换与上式均成立。

\paragraph{考试易错点}
这里必须明确 $a>|b|$：它保证原积分分母无零点，
且单位圆上不出现极点。
复变法应说明极点在单位圆内外的位置；
实变半角代换时不可漏掉跨越 $\theta=\pi$ 的第二段积分。


% 第 12 张板书：考试重点例题的课堂完整留数法推导。
\subsection*{考试重点续讲：老师实际板书的留数计算步骤}
\noindent\textbf{【考试重点｜前一例的课堂板书解法】}
下面依照老师本次板书的次序，详细写出根的选择和 Laurent
系数 $C_{-1}$ 的计算（答案与前一例相同）。

\paragraph{1.\ 单位圆代换}
令 $z=e^{i\theta}$，则
\[
  \cos\theta=\frac{z+z^{-1}}2,\qquad
  \sin\theta=\frac{z-z^{-1}}{2i},
  \qquad dz=iz\,d\theta,\quad d\theta=\frac{dz}{iz}.
\]
$\theta$ 从 $0$ 到 $2\pi$ 时，$z$ 逆时针绕
$|z|=1$ 一周。原题在 $a>|b|\ge0$ 下有
\[
  I_{a,b}=\int_0^{2\pi}\frac{d\theta}{a+b\cos\theta}.
\]
若 $b=0$，直接得 $I_{a,0}=2\pi/a$。
下面设 $b\ne0$：
\begin{align*}
 I_{a,b}
 &=\oint_{|z|=1}
   \frac{1}{a+\frac b2(z+z^{-1})}\frac{dz}{iz}\\
 &=\frac2{i}\oint_{|z|=1}\frac{dz}{bz^2+2az+b}\\
 &=\frac{2}{bi}\oint_{|z|=1}
    \frac{dz}{z^2+\frac{2a}{b}z+1}.
\end{align*}

\paragraph{2.\ 求两个极点并判断其位置}
令 $g(z)=z^2+\frac{2a}{b}z+1$。
因为 $a>|b|$，有
$\Delta:=\sqrt{a^2-b^2}>0$。
求 $g(z)=0$ 得
\[
  z_1=\frac{-a+\Delta}{b}
      =-\frac{b}{a+\Delta},
  \qquad
  z_2=\frac{-a-\Delta}{b}.
\]
由韦达定理，
\[
  z_1z_2=1.
\]
又因为 $a+\Delta>|b|$，
\[
  |z_1|=\frac{|b|}{a+\Delta}<1,
  \qquad
  |z_2|=\frac{1}{|z_1|}>1.
\]
故\textbf{只有 $z_1$ 位于单位圆内部}。
具体而言，$b>0$ 时 $-1<z_1<0$，
$b<0$ 时 $0<z_1<1$；不能在未限定 $b$ 符号
时直接断言两个根的正负。

\paragraph{3.\ 计算单位圆内的一阶极点的留数}
对有理函数 $1/g(z)$，因为
\[
  g'(z)=2z+\frac{2a}{b},
  \qquad
  g'(z_1)=2z_1+\frac{2a}{b}
          =\frac{2\Delta}{b}\ne0,
\]
故 $z_1$ 是一阶极点，Laurent 系数满足
\begin{align*}
 C_{-1}
 &=\operatorname{Res}_{z=z_1}\frac{1}{g(z)}
   =\frac{1}{g'(z_1)}\\
 &=\frac{1}{2z_1+2a/b}
   =\frac{b}{2\sqrt{a^2-b^2}}.
\end{align*}
这正是老师板书上右侧使用的
$C_{-1}=p(z_1)/g'(z_1)$ 公式（这里 $p=1$）。

\paragraph{4.\ 用留数定理写出最终答案}
\begin{align*}
 I_{a,b}
 &=\frac{2}{bi}\cdot 2\pi i\,C_{-1}\\
 &=\frac{2}{bi}\cdot 2\pi i
      \cdot\frac{b}{2\sqrt{a^2-b^2}}\\
 &=\boxed{\frac{2\pi}{\sqrt{a^2-b^2}}}.
\end{align*}
对于 $b=0$，因 $a>0$，
上述结果成为 $2\pi/a$，与直接积分一致。
又由周期性及 $\cos\theta=\sin(\theta+\pi/2)$，
\[
 \boxed{
 \int_0^{2\pi}\frac{d\theta}{a+b\sin\theta}
 =\int_0^{2\pi}\frac{d\theta}{a+b\cos\theta}
 =\frac{2\pi}{\sqrt{a^2-b^2}}
 },\qquad a>|b|.
\]

\paragraph{考试时最应保留的三步}
\textbf{单位圆代换 $\to$ 判定一个极点在圆内
$\to$ 用 $C_{-1}=1/g'(z_1)$ 套留数定理。}
特别注意：即便 $b<0$，只要 $a>|b|$，上述选择
的 $z_1$ 仍然满足 $|z_1|<1$，最后答案不变。


% 第 13 张板书：平方三角函数分母与更高次幂的定积分。
\subsection*{考试重点延伸：含 $\cos^2\theta,\sin^2\theta$ 的积分}

\paragraph{例：求 $\boldsymbol{I_{A,B}}$（接上一题的标准积分）}
设 $A,B\in\mathbb R$ 且 $AB>0$。计算
\[
  I_{A,B}
   =\int_0^{2\pi}
     \frac{d\theta}{A^2\cos^2\theta+B^2\sin^2\theta}.
\]
利用倍角公式
\[
  \cos^2\theta=\frac{1+\cos2\theta}{2},
  \qquad
  \sin^2\theta=\frac{1-\cos2\theta}{2},
\]
可得
\begin{align*}
  I_{A,B}
  &=\int_0^{2\pi}
      \frac{2\,d\theta}
      {A^2(1+\cos2\theta)+B^2(1-\cos2\theta)}\\
  &=\int_0^{2\pi}
      \frac{d(2\theta)}
      {(A^2+B^2)+(A^2-B^2)\cos2\theta}\\
  &=\int_0^{4\pi}
      \frac{d\varphi}
      {(A^2+B^2)+(A^2-B^2)\cos\varphi}\\
  &=2\int_0^{2\pi}
      \frac{d\varphi}
      {(A^2+B^2)+(A^2-B^2)\cos\varphi}.
\end{align*}
这里在第三行令 $\varphi=2\theta$，第四行利用
被积函数的 $2\pi$ 周期性。
沿用上题公式
\[
  \int_0^{2\pi}\frac{d\varphi}{a+b\cos\varphi}
     =\frac{2\pi}{\sqrt{a^2-b^2}}\quad(a>|b|),
\]
取
\[
  a=A^2+B^2,\qquad b=A^2-B^2.
\]
因为 $AB>0$，所以 $A,B$ 均不为零，
$a>|b|$ 且
\[
  \sqrt{a^2-b^2}
  =\sqrt{(A^2+B^2)^2-(A^2-B^2)^2}
  =2|AB|=2AB.
\]
因此
\[
  \boxed{
  I_{A,B}=\frac{4\pi}{2AB}=\frac{2\pi}{AB},
  \qquad AB>0.}
\]
\textbf{注意：}若仅假设 $AB\ne0$ 而不要求同号，
答案应为 $2\pi/|AB|$。

\paragraph{例：带参数 $\boldsymbol{\rho}$ 的三角积分}
按板书写为
\[
  I_{\rho}
   =\int_0^{2\pi}
     \frac{d\theta}{1-\rho\cos\theta+\rho^2},
  \qquad \rho\in\mathbb R.
\]
利用整个周期的正弦、余弦互换，有
\[
  I_{\rho}
  =\int_0^{2\pi}
     \frac{d\theta}{1-\rho\sin\theta+\rho^2}.
\]
令 $a=1+\rho^2$、$b=-\rho$。
对于任意实数 $\rho$，均有
$a>|b|$（因为 $1+\rho^2-|\rho|>0$），故
\begin{align*}
  I_{\rho}
   &=\frac{2\pi}{\sqrt{a^2-b^2}}\\
   &=\frac{2\pi}
      {\sqrt{(1+\rho^2)^2-\rho^2}}\\
   &=\boxed{\frac{2\pi}{\sqrt{1+\rho^2+\rho^4}}}.
\end{align*}
\textbf{核对说明：}对板书所写的分母
$1-\rho\cos\theta+\rho^2$，
严格展开必须是
$(1+\rho^2)^2-\rho^2=1+\rho^2+\rho^4$；
若黑板最末行写成 $1-\rho^2+\rho^4$，
则该行存在合并同类项的笔误。这里按原积分纠正。

\subsection*{更一般的积分：分母的平方与 $n$ 次幂}

\paragraph{老师写出的两个问题}
设 $a,b\in\mathbb R$，$a>|b|$，则
\[
  J_{a,b}
  =\int_0^{2\pi}\frac{d\theta}{(a+b\cos\theta)^2}
  =\int_0^{2\pi}\frac{d\theta}{(a+b\sin\theta)^2}.
\]
更一般地，讨论 $n\in\mathbb N^+$ 时
\[
  J_{a,b}^{(n)}
  =\int_0^{2\pi}\frac{d\theta}{(a+b\cos\theta)^n}
  =\int_0^{2\pi}\frac{d\theta}{(a+b\sin\theta)^n}.
\]
上述等式来自平移变量和三角函数的周期性。

\paragraph{补充解答：对参数 $a$ 求导}
前面已经求得
\[
  J_{a,b}^{(1)}
  =\int_0^{2\pi}\frac{d\theta}{a+b\cos\theta}
  =\frac{2\pi}{\sqrt{a^2-b^2}}.
\]
对固定 $b$，在 $a>|b|$ 内可在积分号下对 $a$
求导。由
\[
  \frac{\partial}{\partial a}
     (a+b\cos\theta)^{-n}
   =-n(a+b\cos\theta)^{-(n+1)},
\]
得到递推关系
\[
  \boxed{J_{a,b}^{(n+1)}
    =-\frac1n\,\frac{\partial}{\partial a}
                   J_{a,b}^{(n)}.}
\]
特别地，
\begin{align*}
  J_{a,b}^{(2)}
  &=-\frac{\partial}{\partial a}
       \left(\frac{2\pi}{\sqrt{a^2-b^2}}\right)\\
  &=\boxed{\frac{2\pi a}{(a^2-b^2)^{3/2}}}.
\end{align*}
继续求导又得
\[
  J_{a,b}^{(3)}
  =\frac{\pi(2a^2+b^2)}{(a^2-b^2)^{5/2}}.
\]
因此，对所有整数 $n\ge1$ 均有
\[
  \boxed{
    J_{a,b}^{(n)}
    =\frac{(-1)^{n-1}}{(n-1)!}
       \frac{\partial^{\,n-1}}{\partial a^{\,n-1}}
       \left(\frac{2\pi}{\sqrt{a^2-b^2}}\right),
    \qquad a>|b|.
  }
\]

% 这里的参数求导法为根据课堂内容补全的解法。
