\section{留数（Residues）}

\begin{proposition}[解析函数零点的孤立性]
若 $f(z)$ 在 $z_0$ 点解析，$f(z_0)=0$，且 $f(z)\not\equiv 0$，
则存在 $\delta>0$，使得在 $|z-z_0|<\delta$ 内有
\[
  f(z)=(z-z_0)^n\varphi(z),
\]
这里 $n\in\mathbb{N}^{+}$，$\varphi(z_0)\ne 0$，且
$\varphi(z)$ 在 $|z-z_0|<\delta$ 内处处可导（解析）。
\end{proposition}

\paragraph{证明}
因 $f(z)$ 在 $z_0$ 解析，故存在 $\delta>0$，当 $|z-z_0|<\delta$ 时，有
\begin{align*}
  f(z)
  &=\sum_{k=0}^{+\infty}\frac{f^{(k)}(z_0)}{k!}(z-z_0)^k \\
  &=f(z_0)+\frac{f'(z_0)}{1!}(z-z_0)+\cdots
    +\frac{f^{(n)}(z_0)}{n!}(z-z_0)^n+\cdots.
\end{align*}
令
\[
  n=\min\bigl\{k: f^{(k)}(z_0)\ne 0\bigr\},
\]
则
\[
  f(z_0)=f'(z_0)=\cdots=f^{(n-1)}(z_0)=0.
\]
于是
\begin{align*}
  f(z)
  &=\frac{f^{(n)}(z_0)}{n!}(z-z_0)^n
    +\frac{f^{(n+1)}(z_0)}{(n+1)!}(z-z_0)^{n+1}+\cdots \\
  &=(z-z_0)^n\underbrace{\left[
      \frac{f^{(n)}(z_0)}{n!}
      +\frac{f^{(n+1)}(z_0)}{(n+1)!}(z-z_0)+\cdots
    \right]}_{\displaystyle\varphi(z)} \\
  &=(z-z_0)^n\varphi(z).
\end{align*}
% 2026-10-19，第 2 张板书：续证命题 1、命题 2。
由于
\[
  \varphi(z)
  =\frac{f^{(n)}(z_0)}{n!}
    +\frac{f^{(n+1)}(z_0)}{(n+1)!}(z-z_0)+\cdots,
  \qquad
  \varphi(z_0)=\frac{f^{(n)}(z_0)}{n!}\ne 0.
\]
又因为 $\varphi(z)$ 在 $z_0$ 解析，所以在 $z_0$ 连续。
由连续函数在一点不为零的性质可知，
存在 $\delta>0$，使得
\[
  \varphi(z)\ne 0,\qquad
  \forall z\in\{z:|z-z_0|<\delta\}.
\]
因此，在该邻域内，由 $f(z)=(z-z_0)^n\varphi(z)$ 可知，
$f(z)=0$ 当且仅当 $z=z_0$。
即 $z_0$ 是 $f(z)$ 在邻域 $|z-z_0|<\delta$ 内的唯一零点。
\hfill$\square$

\begin{proposition}[解析函数唯一性定理]
设 $D$ 为区域，$f(z)$、$g(z)$ 在 $D$ 内解析。
若存在 $D$ 中一列互异点 $z_1,z_2,\ldots,z_n,\ldots$，满足
\[
  z_n\longrightarrow z^*\in D,
  \qquad f(z_n)=g(z_n)\quad(n=1,2,\ldots),
\]
则
\[
  f(z)\equiv g(z),\qquad \forall z\in D.
\]
\end{proposition}

\paragraph{证明}
令
\[
  h(z)=f(z)-g(z).
\]
则 $h(z)$ 在 $D$ 内解析，且
\[
  h(z_1)=h(z_2)=\cdots=h(z_n)=\cdots=0.
\]
由 $z_n\to z^*\in D$ 及 $h(z)$ 在 $z^*$ 处连续，知
\[
  h(z^*)=\lim_{n\to\infty}h(z_n)=0.
\]
若 $f(z)\not\equiv g(z)$，则 $h(z)\not\equiv0$。
根据命题~1（解析函数零点的孤立性），$z^*$ 必为
$h(z)$ 的孤立零点；即存在 $\delta>0$，
在 $0<|z-z^*|<\delta$ 内 $h(z)\ne0$。
但互异的零点列 $z_n\to z^*$，必有某个 $z_n$ 满足
\[
  0<|z_n-z^*|<\delta,\qquad h(z_n)=0,
\]
矛盾。因此 $h(z)\equiv0$，即
\[
  \boxed{f(z)\equiv g(z)\quad(z\in D).}
\]
\hfill$\square$

\subsection*{奇点的分类（Singularities）}

奇点分为\textbf{孤立奇点}与\textbf{非孤立奇点}。
孤立奇点进一步分为：
\[
  \left\{
  \begin{array}{l}
    \text{极点（pole）},\\
    \text{可去奇点（removable singularity）},\\
    \text{本性奇点（essential singularity）}.
  \end{array}
  \right.
\]

\paragraph{非孤立奇点}
例如
\[
  f(z)=\frac{1}{\sin\frac{1}{z}},\qquad z\ne 0.
\]
取
\[
  z_k=\frac{1}{k\pi},\qquad k\in\mathbb Z\setminus\{0\},
\]
则 $z_k$ 是奇点，且 $z_k\to0$（当 $|k|\to\infty$）。
所以 $z_0=0$ 是一个\textbf{非孤立奇点}。

另一个板书例子是
\[
  f(z)=|z|^2=z\overline z.
\]
它在 $z_0=0$ 处复可导，且 $f'(0)=0$；
但在任何包含 $0$ 的邻域内均不解析。
% 注：在一点复可导并不意味着该点解析。

\paragraph{极点（pole）}
设 $z_0$ 为 $f(z)$ 的孤立奇点。
若 $f(z)$ 在 $z_0$ 附近的 Laurent 级数为
\begin{align*}
  f(z)
  &=\frac{C_{-n}}{(z-z_0)^n}
   +\frac{C_{-(n-1)}}{(z-z_0)^{n-1}}
   +\cdots\\
  &\quad +C_0+C_1(z-z_0)+\cdots+C_n(z-z_0)^n+\cdots,
\end{align*}
其中 $C_{-n}\ne0$，$n\in\mathbb N^+$，
则称 $z_0$ 为 $f(z)$ 的\textbf{$n$ 阶极点}。
此时
\[
  \lim_{z\to z_0}f(z)=\infty
  \qquad\bigl(\text{即 }|f(z)|\to\infty\bigr).
\]

\paragraph{可去奇点（removable singularity）}
设 $z_0$ 为 $f(z)$ 的孤立奇点。
若其 Laurent 级数的所有负幂项系数为零，
即 $C_{-n}=0$（$\forall n\ge1$），则
\[
  f(z)=C_0+C_1(z-z_0)+C_2(z-z_0)^2+\cdots,
\]
即在去心邻域中表现为 Taylor 级数，并且
\[
  \lim_{z\to z_0}f(z)=C_0.
\]
在 $z_0$ 处补充定义 $f(z_0)=C_0$，即可使其在 $z_0$ 处解析。
